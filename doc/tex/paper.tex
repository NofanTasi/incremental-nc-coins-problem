\begin{abstract}

\vspace{8pt}

\noindent The \emph{$n$-queens problem} on an $n \!\times\! n$ square allows, on any row, column,
sum-diagonal or diff-diagonal, no more than $1$ queen. The $n = 1$ case has a trivial single
solution, and the $n = 2$ and $n = 3$ cases have no solution, so, in this paper, 
\emph{we limit ourselves to $n \ge 4$}.

\vspace{8pt}

\noindent The \emph{$(n,c)$-coins problem} on an $n \!\times\! n$ square, for $0 \le c \le n$,
generalizes the $n$-queens problem by allowing, on any row, column, sum-diagonal or
difference-diagonal, no more than $c$ coins (for the $n$-queens problem $c = 1$).

\vspace{8pt}

\noindent \emph{We prove the following}:

\vspace{8pt}

\begin{itemize}
\item For every natural number $n \in \mathbb{N}$ with $n \ge 4$, an
\emph{incremental $n \!\times\! n$ Latin square} represents $n + 1$ solutions of
$(n,c)$-coins problems for all $c \in \mathbb{N}$ with $0 \le c \le n$.
\item For every natural number $n \in \mathbb{N}$ with $n \ge 4, \; n \notin \{4, 6\}$, 
there \emph{exists} an incremental \emph{cyclic} $n \!\times\! n$ Latin square.
\end{itemize}

\vspace{8pt}

\noindent \emph{The following is remarkable about AI in general, and Gemini 4 (Argon), in particular}:

\vspace{8pt}

\begin{itemize}
\item The propositions were proved without, as far as we know, any public (published) knowledge
about them, let alone a proof of them.
\item The proofs introduced the \emph{injective-majorization principle}, avoiding intensive
cardinality counting on each diagonal, which would computationally explode for large $n$.
\item The proofs introduced four subsets of $\mathbb{N}$ covering all $n$ except
$n \in \{8, 9, 15, 21\}$:
\begin{itemize}
\item $n \equiv 1,5 \pmod 6$, $n \ge 5$,
\item $n \equiv 2,4 \pmod 6$, $n \ge 10$,
\item $n \equiv 3 \pmod 6$, $n \ge 27$,
\item $n \equiv 0 \pmod 6$, $n \ge 12$.
\end{itemize}
\end{itemize}

\vspace{8pt}

\noindent For every subset above, a cyclic Latin square can be constructed \emph{collectively} for
all its elements.

\vspace{8pt}

\noindent Constructing a cyclic Latin square for the four uncovered natural numbers is handled
\emph{individually}.

\vspace{8pt}

\noindent See \href{https://github.com/NofanTasi/incremental-nc-coins-problem}{GitHub}
and also
\noindent
\href{https://mathoverflow.net/questions/510135}{MO\hspace{-0.5pt}\#\hspace{-0.5pt}510135},

\section{Assumptions, Notation and Definitions}

In what follows, $n \in \mathbb{N}$ is a natural number with $n \ge 4$.

\vspace{8pt}

For all $m \le n$, we write $[m,n]$ for $\{m, \dots, n\}$ (in particular, $[0,n]$ for
$\{0, \dots, n\}$).

\vspace{8pt}

In what follows, $S^n : [0,n-1]^2 \to [0,n-1]$ is referred to as a square.

\vspace{8pt}

When dealing with squares and related concepts, like rows, columns, diagonals and cells,
it is important to distinguish them from their values. 

\vspace{8pt}

For squares and related concepts we use \emph{uppercase} notation, whereas for their values
we use corresponding \emph{lowercase} notation. For example, we write $s^n_{i,j}$ for the 
value of the $(i,j)$-cell, $S^n_{i,j}$, of square $S^n$.

\vspace{8pt}

We write $S^n_{i,j}$ ($s^n_{i,j}$) for (the value of) the $(i,j)$-cell of $S^n$,
$\forall i,j \in [0,n-1]$.

\vspace{8pt}

We write $R^n_i$ ($r^n_{i,j}$) for (the values of) $\{S^n_{i,j} \mid j \in [0,n-1]\}$, the $i$-row
of  $S^n$, $\forall i \in [0,n-1]$.

\vspace{8pt}

We write $C^n_j$ ($c^n_{i,j}$) for (the values of) $\{S^n_{i,j} \mid i \in [0,n-1]\}$, the $j$-column
of $S^n$, $\forall j \in [0,n-1]$.

\vspace{8pt}

We write $D^n_{i+j=s}$ ($d^n_{i+j=s}$) for (the values of) 
$\{S^n_{i,j} \mid i,j \in [0,n-1], \; i + j = s\}$, the $s$-sum-diagonal, also simply called
$s$-diagonal, of $S^n$, $\forall s \in [0,2(n-1)]$.

\vspace{8pt}

We write $D^n_{i-j=d}$ ($d^n_{i-j=d}$) for (the values of)
$\{S^n_{i,j} \mid i,j \in [0,n-1], \; i - j = d\}$, the $d$-diff-diagonal, also simply called
$d$-diagonal, of $S^n$, $\forall d \in [-(n-1),n-1]$.

\vspace{8pt}

We write $D^{n,k}_{i+j=s}$ for $\{S^n_{i,j} \mid S^n_{i,j} \in D^n_{i+j=s}, \; s^n_{i,j} \le k\}$, 
$\forall s \in [0,2(n-1)] \, , \, \forall k \in [0,n-1]$.

\vspace{8pt}

We write $D^{n,k}_{i-j=d}$ for $\{S^n_{i,j} \mid S^n_{i,j} \in D^n_{i-j=d}, \; s^n_{i,j} \le k\}$, 
$\forall d \in [-(n-1),n-1] \, , \, \forall k \in [0,n-1]$.

\vspace{8pt}

\begin{definition}\label{def:nc}
\hspace{1pt} \\ A solution of an $(n,c)$-coins problem, for $c \in [0,n]$, is a subset $S^n_c$ of
$n\,.\,c$ cells of a square $S^n$ such that every row $R^n_i$, column $C^n_j$,
diagonal $D^n_{i+j=s}$, and diagonal $D^n_{i-j=d}$, contains at most $c$ cells of $S^n_c$.
\end{definition}

\vspace{8pt}

Le, Li and Wang~\cite{LLW1990} construct solutions of $(n,c)$-coins problems \emph{separately} for
each $c \in [0,n]$.

\vspace{8pt}

\begin{definition}\label{def:is}
\hspace{1pt} \\ A square $S^n$ is an \emph{incremental} square $\Leftrightarrow$
\begin{itemize}
\item $|\,D^{n,k}_{i+j=s}\,| \le k + 1$, 
$\forall s \in [0,2(n-1)] \, , \, \forall k \in [0,n-1]$,
\item $|\,D^{n,k}_{i-j=d}\,| \le k + 1$, 
$\forall d \in [-(n-1),n-1] \, , \, \forall k \in [0,n-1]$.
\end{itemize}
\end{definition}

\vspace{8pt}

\begin{definition}\label{def:ls}
\hspace{1pt} \\ A square $S^n$ is a \emph{Latin} square $\Leftrightarrow$ 
\begin{itemize}
\item $r^n_i$ is a permutation of $[0,n-1]$, $\, \forall i \in [0,n-1] \;$,
\item $c^n_j$ is a permutation of $[0,n-1]$, $\, \forall j \in [0,n-1] \;$.
\end{itemize}
\end{definition}

\vspace{8pt}

For a Latin square, we write $L^{n,p}$ for $S^{n,p}$ and $l^{n,p}_{i,j}$ for $s^{n,p}_{i,j}$.

\vspace{8pt}

\begin{definition}\label{def:cs}
\hspace{1pt} \\ A square $S^n$ is a \emph{cyclic} square $\Leftrightarrow$
$s^n_{i + 1, j} = s^n_{i,j} + 1 \bmod n, \; \forall i,j \in [0,n-1]$.
\end{definition}

\vspace{12pt}

In what follows, $p$ is a permutation of $[0,n-1]$.

\vspace{8pt}

We write $S^{n,p}$ for the square with cell values 
$s^{n,p}_{i,j} = (i + p(j)) \bmod n, \; \forall i,j \in [0,n-1]$.

\vspace{8pt}

Clearly, $S^{n,p}$ is a cyclic Latin square, so we use notation $L^{n,p}$ and $l^{n,p}_{i,j}$.

\section{Theorem Statements}

\begin{theorem}\label{thm:main1}
\hspace{1pt} \\ $\forall n \in \mathbb{N}, \; n \ge 4$, an incremental Latin $n \!\times\! n$ square
represents $n + 1$ solutions of $(n,c)$-coins problems \emph{simultaneously} for $c \in [0,n]$ as
$S^n_c = \{L^n_{i,j} \mid l^n_{i,j} < c\}$ (notice that $S^n_0 = \emptyset$).
\end{theorem}

\vspace{8pt}

\begin{theorem}\label{thm:main2}
\hspace{1pt} \\ $\forall n \in \mathbb{N}, \; n \ge 4, \; n \notin \{4, 6\} \; \exists L^{n,p}$ that
is an \emph{incremental} cyclic Latin square.
\end{theorem}

\vspace{8pt}

Notice that the first theorem is not constructive, whereas the second theorem is constructive.

\section{Injective Majorization}

Let $L^{n,p}$ be a cyclic Latin square as above and let $D$ be any diagonal.

\vspace{8pt}

Instead of using cardinality counting, which would computationally explode for large $n$, the
\emph{injective majorization principle} reduces it, for \emph{all} thresholds $k \in [0,n-1]$,
simultaneously to construct, for each diagonal direction ($D = D^n_{i+j=s}$ or $D = D^n_{i-j=d}$),
to a single majorized injection, for \emph{all} diagonals involved.

\vspace{8pt} 

\begin{lemma}\label{lem:inj-maj}
If there exists a function $f : [0,n-1]^2 \to \mathbb{N}$ such that:

\vspace{8pt}

\begin{enumerate}[nosep,label=(\arabic*)]
  \item Majorization: $\forall i,j \in [0,n-1], f(i,j) \le l^{n,p}_{i,j}$ on $D$.
  \item Injectivity: $f$ is injective on $D$.
\end{enumerate}

\vspace{8pt}

Then for every threshold $k \in [0,n-1]$, and every diagonal
$D^k$ ($D^{n,k}_{i+j=s}$ or $D^{n,k}_{i-j=d}$), $|\,D^k\,| \le k + 1$.
\end{lemma}

\begin{proof}

\vspace{8pt}

\hspace{1pt} \\ By condition~(1), $0 \le f(i,j) \le l^{n,p}_{i,j} \le k$ on $D^k$,
therefore $f(i,j) \in [0,k]$.
\\ By condition~(2), the restriction of $f$ to $D^k$ is an injective map into $[0,k]$,
therefore $|D^k| \le |\,[0,k]\,| = k + 1$.

\end{proof}
