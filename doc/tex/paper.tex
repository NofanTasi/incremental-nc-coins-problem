\title{\textbf{
  $n \times n$ Cyclic Latin Squares \\ 
  representing solutions of the \\
  $(n,c)$-Coins Problem for $0 \le c \le n$
}}
\author{Marc Duponcheel and Luc Duponcheel}

\date{\today}

\begin{document}

\maketitle

\begin{abstract}

\vspace{8pt}
 
\noindent $\forall n \in \mathbb{N}, \; n > 0, \; n \notin \{2, 3, 4, 6\}$, 
$\exists p: \{ 0, \ldots n-1 \} \to \{ 0, \ldots n-1 \}$, a permutation of $\{ 0, \ldots n-1 \}$, 
such that $\forall i, j \in \{ 0, \ldots n-1 \}$, the square with cells
$S_{i,j} = (i + p(j)) \bmod n$ is an $n \times n$ \emph{Cyclic Latin Square}, representing solutions
of $(n,c)$\emph{-Coins Problems} $\forall c \;\; 0 \le c \le n$. An $(n,c)$-Coins Problem for
$c = 1$ corresponds to an $n$-\emph{Queens Problem}.

\vspace{8pt}

\noindent We have an \emph{existence proof} in {\tt Lean} and corresponding \emph{construction code}
in {\tt Haskell}, {\tt Julia} and {\tt Lean}.

\vspace{8pt}

\noindent
See \href{https://github.com/NofanTasi/incremental-nc-coins-problem}{GitHub}
and also
\noindent
\href{https://mathoverflow.net/questions/510135}{MO\hspace{-0.5pt}\#\hspace{-0.5pt}510135},
\href{https://puzzling.stackexchange.com/questions/138939}%
{PSE\hspace{-0.5pt}\#\hspace{-0.5pt}138939},
\href{https://mathoverflow.net/questions/516225}{MO\hspace{-0.5pt}\#\hspace{-0.5pt}516225}.

\vspace{8pt}

\noindent The proof uses the \emph{Injective-Majorization Principle} that, instead of intensive
checking inequalities on each diagonal, uses the existence of a single function
$f(i,j) \le S_{i,j}$ that is injective on each diagonal. 

\vspace{8pt}

\noindent The proof uses four subsets of $\mathbb{N}$ covering almost all 
$\mathbb{N}\,mod\,6$ classes~:

\vspace{8pt}

\begin{itemize}
\item $n \equiv 1,5 \pmod 6$ (with $n \ge 1$),
\item $n \equiv 2,4 \pmod 6$ (with $n \ge 10$),
\item $n \equiv 3 \pmod 6$ (with $n \ge 27$),
\item $n \equiv 0 \pmod 6$ (with $n \ge 12$).
\end{itemize}

\vspace{8pt}

\noindent For all four subsets a Cyclic Latin Square can be constructed collectively for all their
elements.

\vspace{8pt}

\noindent Constructing a Cyclic Latin Square for the four uncovered natural numbers,
$n \in \{8, 9, 15, 21\}$, is handled individually. 

\vspace{8pt}

\noindent Likewise, constructing a (non-Cyclic) Latin Square for the two remaining natural numbers, 
$n \in \{4, 6\}$, is also handled individually.

\end{abstract}

\section{Cyclic Latin Squares}

\subsection{From \texorpdfstring{$(n,c)$}{(n,c)}-Coins Problems to Latin Squares}

Given $n > 0$, and $c \in \{0, \dots, n\}$ solutions of $(n,c)$-Coins Problems are subsets $S_c$ of
$n.c$ cells of an $n \times n$ square $S_{i,j}$ such that every row, column,
$s$-diagonal ($i + j = s$), and $d$-diagonal ($i - j = d$), contains at most $c$ cells of $S_c$.

\vspace{8pt}

The construction of Le, Li and Wang~\cite{LLW1990} constructs \emph{one} $S_c$ for each $c$. 

\vspace{8pt}

Our construction constructs an $n \times n$ Cyclic Latin Square $S_{i,j}$ representing \emph{many}
$S_c$'s at once using \\ 
$S_c = \{(i,j) \mid S_{i,j} \le c\} \;\; \forall c \;\; 0 \le c \le n$.

\section{Theorem Statement}

\begin{theorem}\label{thm:main}
$\forall n \in \mathbb{N}, \; n > 0, \; n \notin \{2, 3, 4, 6\}$,
$\exists p: \{ 0, \ldots n-1 \} \to \{ 0, \ldots n-1 \}$, a permutation, such that 
$S_{i,j} = (i + p(j)) \bmod n$ is a Cyclic Latin
Square satisfying~:

\vspace{8pt}

\begin{enumerate}[nosep,label=(\roman*)]
  \item $\forall i \in \{ 0, \ldots n-1 \}$, the row map $j \mapsto S_{i,j}$ is bijective.
  \item $\forall j \in \{ 0, \ldots n-1 \}$, the column map $i \mapsto S_{i,j}$ is bijective.
  \item $\forall k : 0 \le k < n, \forall s \in \mathbb{N}$,
        $|\{(i,j) \in$ $s$-diagonal with $S_{i,j} \le k\}| \le k + 1$.
  \item $\forall k : 0 \le k < n, \forall d \in \mathbb{Z}$,
        $|\{(i,j) \in$ $d$-diagonal with $S_{i,j} \le k\}| \le k + 1$.
\end{enumerate}
\end{theorem}

\section{Injective Majorization}

Instead of using direct cardinality counting for $(iii)$ and $(iv)$ above,  which would
computationally explode for large $n$, we use a principle that reduces the problem for \emph{all}
thresholds $0 \le k < n$ simultaneously to constructing a single dominated injection.

\vspace{8pt}

In what follows, unless explicitely stated, diagonal refers to either $s$-diagonal or $d$-diagonal.

\subsection{The Injective-Majorization Principle}

Let $S_{i,j}$ with $0 \le i, j < n$ be a square and let $D$ be any diagonal.

\begin{lemma}\label{lem:inj-maj}
If there exists a function $f : \mathbb{N} \times \mathbb{N} \to \mathbb{N}$ such that:

\vspace{8pt}

\begin{enumerate}[nosep,label=(\arabic*)]
  \item Majorization~: $\forall i, j : 0 \le i, j < n, f(i,j) \le S_{i,j}$.
  \item Diagonal Injectivity~: $f$ is injective on $D$.
\end{enumerate}

\vspace{8pt}

Then for every threshold $k \in \mathbb{N}$, the set
$D_{\le k} = \{(i,j) \in D \mid S_{i,j} \le k\}$ has cardinality at most $k + 1$.
\end{lemma}

\begin{proof}

\vspace{8pt}

By condition~(1), $0 \le f(i,j) \le S_{i,j} \le k$ so  $f(i,j) \in \{0, \dots, k\}$.
By condition~(2), the restriction of $f$ to $D_{\le k}$ is an injective map into $\{0, \dots, k\}$.
Therefore $|D_{\le k}| \le |\{0, \dots, k\}| = k + 1$.

\end{proof}

\section{Theorem Proof}

Recall that $i = s - j$ on every $s$-diagonal and $i = d + j$ on every $d$-diagonal. 

\vspace{8pt}

The explanations below will use those equalities without explicitly mentioning them.

\vspace{8pt}

Recall that the four families are

\begin{itemize}
\item Family 1~: $n \equiv 1,5 \pmod 6$ with $n \ge 1$,
\item Family 2~: $n \equiv 2,4 \pmod 6$ with $n \ge 10$, 
\item Family 3~: $n \equiv 3 \pmod 6$ with $n \ge 27$,
\item Family 4~: $n \equiv 0 \pmod 6$ with $n \ge 12$.
\end{itemize}

and notice that

\begin{itemize}
\item Family 2 does not include $n = 8$.
\item Family 3 does not include $n = 9$, $n = 15$ and $n = 21$.
\end{itemize}

Those cases are handled separately.

\vspace{8pt}

We will define $p$ and $f$ and prove that

\begin{itemize}
\item $p$ is a bijection on $\{0, \dots, n-1\}$,
\item $f$ is an injection on the diagonals involved,
\item $f(i, j) \le S_ij$ on the diagonals involved.
\end{itemize}

\subsection{Family 1}

Let $p(j) = (-2j) \bmod n$.

\vspace{8pt}

$p$ is bijective since $\gcd(n, 2) = 1$.

\vspace{8pt}

On an $s$-diagonal, $S_{i,j} = (s - 3j) \bmod n$ is injective since $\gcd(3, n) = 1$
(see Example~\ref{ex:fam1_sum} and Figure~\ref{fig:fam1_sum}).

\vspace{8pt}

On a $d$-diagonal, $S_{i,j} = (d - j) \bmod n$ is trivially injective
(see Example~\ref{ex:fam1_diff} and Figure~\ref{fig:fam1_diff}).

\vspace{8pt}

Let $f(i,j) = S_{i,j}$.

\subsection{Family 2}

\vspace{8pt}

Let $m = n/2$.

\vspace{8pt}

Let $p(j) = (2j + m + 1) \bmod n$ for $0 \le j < m$. \\
Let $p(j) = (2(j - m) + m + 4) \bmod n$ for $m \le j < n$.

\vspace{8pt}

$p$ is bijective since the former covers all $(m + 1) \bmod 2$ values one-to-one , and the latter
covers all $m \bmod 2$ values one-to-one.

\vspace{8pt}

On an $s$-diagonal, inside each half, $S_{i,j} = (s + j + m + 1) \bmod n$ has step $+1$. Because the
second half is shifted by $3 \bmod n$, a collision between $j_1 < m \le j_2$ happens when 
$j_2 - j_1 = n - 3$. This happens only when $j_1 \in \{0, 1, 2\}$ 
(and $i_1 \in \{n - 3, n - 2, n - 1\}$), the $3 \times 3$ bottom-left corner,
and $j_2 \in \{n - 3, n - 2, n - 1\}$ (and $i_2 \in \{0, 1, 2\}$), the $3 \times 3$ top-right
corner.

\vspace{8pt}

Let $f(i,j) = j - (n - 3) \in \{0, 1, 2\}$ on the $3 \times 3$ top-right corner. One of the values,
skipped by the $+3$ shift, are vacant on each $s$-diagonal passing through it
(see Example~\ref{ex:fam2_sum} and Figure~\ref{fig:fam2_sum}). Trivially $f(i, j) \le S_{i,j}$.

\vspace{8pt}

In the $3 \times 3$ top-right corner ($j = n - 3 + \delta$ with $\delta \in \{0, 1, 2\}$),
$p(j) = m - 2 + 2\delta$, so the largest sum $i + p(j)$ occurs at $(i, j) = (2, n - 1)$, where
$2 + p(n - 1) = m + 4$. The threshold $n \ge 10$ ($m \ge 5$) ensures $m + 4 < 2m = n$, so $S_{i,j}$
never wraps around $\bmod\, n$.

\vspace{8pt}

On a $d$-diagonal, inside each half, $S_{i,j} = s + 3j - m + 4$ has step $+3$.
Because the second half is shifted by $3 \bmod n$, a collision between $j_1 < m \le j_2$
happens when $j_2 - j_1 = n - 1$, equivalent with $(j_2 - j_1) + 1 \equiv 0 \pmod n$, equivalent
with $3(j_2 - j_1) + 3 \equiv 0 \pmod n$ since $\gcd(3, n) = 1$. This happens only at a single pair
of cells on the entire board: $(0, 0)$ and $(n - 1, n - 1)$ on the main diagonal.

\vspace{8pt}

Let $f(n - 1,n - 1) = 1$, one of the values, skipped by the $+3$ shift, which is vacant on the main
diagonal (see Example~\ref{ex:fam2_diff} and Figure~\ref{fig:fam2_diff}).

\subsection{Family 3}

\vspace{8pt}

Let $k = n / 3$.

\vspace{8pt}

Let $p(j) = (3j + 2k + 4u) \bmod n$ with $u = \lfloor j / k \rfloor \in \{0, 1, 2\}$.
$\{0, \dots, n - 1\}$ is partitioned into three blocks of length $k$ with $+4$ jumps between
consecutive blocks.

\vspace{8pt}

$p$ is bijective since block $u$ covers residue class $(2k + u) \bmod 3$, and thus all three
residue classes $\bmod\, 3$ are, one-to-one, covered.

\vspace{8pt}

On an $s$-diagonal, $S_{i,j} = (s + 2j + 2k + 4u) \bmod n$ has step $+2$ inside each block. At each
of the two block transitions ($u = 0 \to 1$ and $u = 1 \to 2$), the extra $+4$ jump skips two steps
of $+2$: $(s + k) \bmod n$ and $(s + k + 2) \bmod n$ at $j = k$, and $(s + 4) \bmod n$ and 
$(s + 6) \bmod n$ at $j = 2k$. Collisions between blocks $u_1 < u_2$ happen
when $2(j_2 - j_1) + 4(u_2 - u_1) \equiv 0 \pmod n$, which is equivalent with
$(j_2 - j_1) + 2(u_2 - u_1) = n$ since $\gcd(2, n) = 1$, forcing $u_1 = 0$, $u_2 = 2$ and
$j_2 - j_1 = n - 4$. This happens only in the $4 \times 4$ top-right corner.

\vspace{8pt}

Let $f(i,j) = S_{i,j} - k \in \{s + k, s + k + 2\} \bmod n$ for $j \in \{n - 4, n - 3\}$, which
fills the two vacant slots skipped at the $u = 0 \to 1$ transition ($j = k$)
(see Example~\ref{ex:fam3_sum} and Figure~\ref{fig:fam3_sum}). \\
Let $f(i,j) = S_{i,j} - 2k \in \{s + 4, s + 6\} \bmod n$ for $j \in \{n - 2, n - 1\}$, which
fills the two vacant slots skipped at the $u = 1 \to 2$ transition ($j = 2k$)
(see Example~\ref{ex:fam3_sum} and Figure~\ref{fig:fam3_sum}).

\vspace{8pt}

In the $4 \times 4$ top-right corner, the largest unreduced sum $i + p(j)$ occurs at
$(i, j) = (3, n - 1)$, where $p(n - 1) = 2k + 5$ and $3 + p(n - 1) = 2k + 8$. The threshold
$n \ge 27$ ($k \ge 9$) ensures $2k + 8 < 3k = n$, so $S_{3, n-1} = 2k + 8$ does not wrap around
$\bmod\, n$ to a value $< 2k$ before subtracting $2k$ in $f(3, n - 1) = S_{3, n-1} - 2k$.

\vspace{8pt}

On a $d$-diagonal, $S_{i,j} = (d + 4j + 2k + 4u) \bmod n$ has step $+4$ inside each
block. At each of the two block transitions ($u = 0 \to 1$ and $u = 1 \to 2$), the extra $+4$ jump
skips one step of $+4$: $d \bmod n$ at $j = k$, and $(d + k + 4) \bmod n$ at $j = 2k$. Collisions
between blocks $u_1 < u_2$ happen when $4(j_2 - j_1) + 4(u_2 - u_1) \equiv 0 \pmod n$, which is
equivalent with $(j_2 - j_1) + (u_2 - u_1) = n$ since $\gcd(4, n) = 1$, forcing $u_1 = 0$,
$u_2 = 2$ and $j_2 - j_1 = n - 2$. This happens only in the $2 \times 2$ bottom-right corner.

\vspace{8pt}

Let $f(i,j) = S_{i,j} - 2k = i + 2 \equiv d \pmod n$ for $j = n - 2$, which fills the vacant
slot skipped at the $u = 0 \to 1$ transition ($j = k$)
(see Example~\ref{ex:fam3_diff} and Figure~\ref{fig:fam3_diff}). \\
Let $f(i,j) = S_{i,j} - k = i + k + 5 \equiv d + k + 4 \pmod n$ for $j = n - 1$, which fills the
vacant slot skipped at the $u = 1 \to 2$ transition ($j = 2k$)
(see Example~\ref{ex:fam3_diff} and Figure~\ref{fig:fam3_diff}).

\subsection{Family 4}

\vspace{8pt}

Let $m = n / 2$ and $k = n / 3$.

\vspace{8pt}

Let $p(0) = n - 2$, $p(m - 2) = 0$ and $p(n - 1) = 2$. \\
Let $p(j) = 2j + 2$ for $1 \le j \le m - 3$. \\
Let $p(j) = 2(j - (m - 1)) + 1$ for $m - 1 \le j \le n - 2$.

\vspace{8pt}

$p$ is bijective since the first ones cover $\{0, 2, n - 2\}$, and the next ones cover
$\{4, 6, \dots, n - 4\}$, and the last ones cover all odd numbers $\{1, 3, \dots, n - 1\}$

\vspace{8pt} 

The threshold $n \ge 12$ ($m \ge 6$ and $k \ge 4$) covers all valid multiples of $6$ since $n = 6$
is excluded, and ensures that the first main block $1 \le j \le m - 3$ is non-empty and that the
column $j = n - 5$ lies inside the second main block $m - 1 \le j \le n - 2$.

\vspace{8pt}

On an $s$-diagonal, inside the first main block $S_{i,j} = (s + j + 2) \bmod n$ has
step $+1$, and inside the second main block $S_{i,j} = (s + j + 3) \bmod n$ also has step $+1$,
leaving the two values $(s + m) \bmod n$ and $(s + m + 1) \bmod n$ vacant between the two blocks.
Collisions only occur with the three boundary pins $j \in \{0, m - 2, n - 1\}$.

\vspace{8pt}

At $j = m - 1$ (for $i < n - 1$): $S_{i, m-1} = i + 1$ collides with pin $j = m - 2$
($S_{i+1, m-2} = i + 1$), so let $f(i, m - 1) = S_{i, m-1} - 1 = i \equiv s + m + 1 \pmod n$
(see Example~\ref{ex:fam4_sum} and Figure~\ref{fig:fam4_sum}). \\
At $j = n - 5$ (for $i + j < n$, so $i < 5$): $S_{i, n-5} = s - 2$ collides with pin $j = 0$
($S_{s, 0} = s - 2$), so let $f(i, n - 5) = S_{i, n-5} - (m - 2) = i + m - 5 \equiv s + m \pmod n$
(see Example~\ref{ex:fam4_sum} and Figure~\ref{fig:fam4_sum}). \\
At $j = n - 1$ (for $i < 2$): $S_{i, n-1} = i + 2$ collides with $j = 1$
($S_{s-1, 1} = i + 2$), while pin $j = 0$ is off the diagonal ($s \ge n - 1$), leaving $i + 1$
vacant; so let $f(i, n - 1) = S_{i, n-1} - 1 = i + 1$
(see Example~\ref{ex:fam4_sum} and Figure~\ref{fig:fam4_sum}).

\vspace{8pt}

On a $d$-diagonal, inside both main blocks $i + p(j)$, $S_{i,j} = (d + 3j + 2) \bmod n$ has step
$+3$: $S_{i,j} = (d + 3j + 2) \bmod n$ in the first main block and $S_{i,j} = (d + 3j + 3) \bmod n$ in
the second main block. Since $3 \mid n$ ($\gcd(3, n) = 3$), the first main block stays entirely in
residue class $(d + 2) \bmod 3$ and the second main block stays entirely in residue class
$d \bmod 3$, leaving the third residue class $(d + 1) \bmod 3$ completely vacant (except for the
single pin $j = 0$). Within each block, a wrap-around collision $3(j_2 - j_1) = n$ occurs when
$j_2 - j_1 = n / 3 = k$ (for $i_2 \ge k$). To resolve these collisions
(see Example~\ref{ex:fam4_diff} and Figure~\ref{fig:fam4_diff}):
\begin{itemize}[nosep]
\item In the first main block ($1 + k \le j \le m - 3$ and $i \ge k$), let
$f(i, j) = S_{i,j} - 1 \equiv d + 1 \pmod 3$.
\item In the second main block ($m + k \le j \le n - 2$ and $i \ge k$), let
$f(i, j) = S_{i,j} - 2 \equiv d + 1 \pmod 3$.
\item At the boundary column $j = m - 1 + k$, let $f(i, j) = S_{i,j} - 1$ for $k \le i < 2k$, and
$f(i, j) = S_{i,j} - 5$ (or $S_{i,j} - 4$ when $n = 12$) for $i \ge 2k$ to avoid colliding with the
pins $j = m - 2$ and $j = 0$.
\end{itemize}

\vspace{8pt}

\begin{example}[Family 3 $s$-Diagonal Block Jumps: $n = 27$, $s = 25$]\label{ex:fam3_sum}
For $n = 27$ ($k = 9$), along $i + j = 25$ the step inside each block is $+2$. At the block jump
$j = 8 \to 9$ ($u = 0 \to 1$), $S_{i,j}$ jumps from $S_{17,8} = 5$ to $S_{16,9} = 11$, skipping
$\mathbf{7}$ and $\mathbf{9}$ ($\equiv s + k, s + k + 2 \pmod{27}$). At $j = 17 \to 18$
($u = 1 \to 2$), $S_{i,j}$ jumps from $S_{8,17} = 0$ to $S_{7,18} = 6$, skipping $\mathbf{2}$
and $\mathbf{4}$ ($\equiv s + 4, s + 6 \pmod{27}$). In the $4 \times 4$ top-right corner
($j_2 - j_1 = n - 4 = 23$), cells $(2, 23)$, $(1, 24)$, $(0, 25)$ collide with $(25, 0)$,
$(24, 1)$, $(23, 2)$ at values $16, 18, 20$. Subtracting $k = 9$ at $j \in \{23, 24\}$ gives
$f(2, 23) = 7$ and $f(1, 24) = 9$, and subtracting $2k = 18$ at $j = 25$ gives $f(0, 25) = 2$,
filling the exact skipped slots (Figure~\ref{fig:fam3_sum}).
\end{example}
