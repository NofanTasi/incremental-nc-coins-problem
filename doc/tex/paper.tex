\begin{abstract}

\vspace{8pt}

\noindent The \emph{$n$-queens problem} on an $n \!\times\! n$ square allows, on any row, column,
sum-diagonal or diff-diagonal, no more than $1$ queen. The $n = 1$ case has a trivial single
solution, and the $n = 2$ and $n = 3$ cases have no solution, so, in this paper, 
\emph{we limit ourselves to $n \ge 4$}.

\vspace{8pt}

\noindent The \emph{$(n,c)$-coins problem} on an $n \!\times\! n$ square, for $0 \le c \le n$,
generalizes the $n$-queens problem by allowing, on any row, column, sum-diagonal or
difference-diagonal, no more than $c$ coins (for the $n$-queens problem $c = 1$).

\vspace{8pt}

\noindent \emph{We prove the following}:

\vspace{8pt}

\begin{itemize}
\item For every natural number $n \in \mathbb{N}$ with $n \ge 4$, an
\emph{incremental $n \!\times\! n$ Latin square} represents $n + 1$ solutions of
$(n,c)$-coins problems for all $c \in \mathbb{N}$ with $0 \le c \le n$.
\item For every natural number $n \in \mathbb{N}$ with $n \ge 4, \; n \notin \{4, 6\}$, 
there \emph{exists} an incremental \emph{cyclic} $n \!\times\! n$ Latin square.
\item For $n \in \{4, 6\}$, there \emph{exists} a non-cyclic incremental $n \!\times\! n$ Latin
square, and hence for every natural number $n \in \mathbb{N}$ with $n \ge 4$, there \emph{exists} an
incremental $n \!\times\! n$ Latin square representing $n + 1$ solutions of $(n,c)$-coins problems
for all $c \in \mathbb{N}$ with $0 \le c \le n$.
\end{itemize}

\vspace{8pt}

\noindent \emph{The following is remarkable about AI in general, and Gemini 4 (Argon), in particular}:

\vspace{8pt}

\begin{itemize}
\item The propositions were proved without, as far as we know, any public (published) knowledge
about them, let alone a proof of them.
\item The proofs introduced the \emph{injective-majorization principle}, avoiding intensive
cardinality counting on each diagonal, which would computationally explode for large $n$.
\item The proofs introduced four subsets of $\mathbb{N}$ covering all $n$ except
$n \in \{8, 9, 15, 21\}$:
\begin{itemize}
\item $n = 1,5 \bmod 6$, $n \ge 5$,
\item $n = 2,4 \bmod 6$, $n \ge 10$,
\item $n = 3 \bmod 6$, $n \ge 27$,
\item $n = 0 \bmod 6$, $n \ge 12$.
\end{itemize}
\end{itemize}

\vspace{8pt}

\noindent For every subset above, a cyclic Latin square can be constructed \emph{collectively} for
all its elements.

\vspace{8pt}

\noindent Constructing a cyclic Latin square for the four uncovered natural numbers is handled
\emph{individually}.

\section{Assumptions, Notation and Definitions}

In what follows, $n \in \mathbb{N}$ is a natural number with $n \ge 4$.

\vspace{8pt}

For all $m \le n$, we write $[m,n]$ for $\{m, \dots, n\}$ (in particular, $[0,n]$ for
$\{0, \dots, n\}$).

\vspace{8pt}

In what follows, $S^n : [0,n-1]^2$ is referred to as a square.

\vspace{8pt}

When dealing with squares and related concepts, like rows, columns, diagonals and cells,
it is important to distinguish them from their values. 

\vspace{8pt}

For squares and related concepts we use \emph{uppercase} notation, whereas for their values
we use corresponding \emph{lowercase} notation. For example, we write $s^n_{i,j}$ for the 
value of the $(i,j)$-cell, $S^n_{i,j}$, of square $S^n$.

\vspace{8pt}

We write $S^n_{i,j}$ ($s^n_{i,j}$) for (the value of) the $(i,j)$-cell of $S^n$,
$\forall i,j \in [0,n-1]$. Notice that we work $0$ based.

\vspace{8pt}

We write $R^n_i$ ($r^n_{i,j}$) for (the values of) $\{S^n_{i,j} \mid j \in [0,n-1]\}$, the $i$-row
of $S^n$, $\forall i \in [0,n-1]$.

\vspace{8pt}

We write $C^n_j$ ($c^n_{i,j}$) for (the values of) $\{S^n_{i,j} \mid i \in [0,n-1]\}$, the $j$-column
of $S^n$, $\forall j \in [0,n-1]$.

\vspace{8pt}

We write $D^n_{i+j=s}$ ($d^n_{i+j=s}$) for (the values of) 
$\{S^n_{i,j} \mid i,j \in [0,n-1], \; i + j = s\}$, the $s$-sum-diagonal, also simply called
$s$-diagonal, of $S^n$, $\forall s \in [0,2(n-1)]$.

\vspace{8pt}

We write $D^n_{i-j=d}$ ($d^n_{i-j=d}$) for (the values of)
$\{S^n_{i,j} \mid i,j \in [0,n-1], \; i - j = d\}$, the $d$-diff-diagonal, also simply called
$d$-diagonal, of $S^n$, $\forall d \in [-(n-1),n-1]$.

\vspace{8pt}

We write $D^{n,k}_{i+j=s}$ for $\{S^n_{i,j} \mid S^n_{i,j} \in D^n_{i+j=s}, \; s^n_{i,j} \le k\}$, 
$\forall s \in [0,2(n-1)] \, , \, \forall k \in [0,n-1]$.

\vspace{8pt}

We write $D^{n,k}_{i-j=d}$ for $\{S^n_{i,j} \mid S^n_{i,j} \in D^n_{i-j=d}, \; s^n_{i,j} \le k\}$, 
$\forall d \in [-(n-1),n-1] \, , \, \forall k \in [0,n-1]$.

\vspace{8pt}

\begin{definition}\label{def:nc}
\hspace{1pt} \\ A solution of an $(n,c)$-coins problem, for $c \in [0,n]$, is a subset $S^n_c$ of
$n\,.\,c$ cells of a square $S^n$ such that every row $R^n_i$, column $C^n_j$,
diagonal $D^n_{i+j=s}$, and diagonal $D^n_{i-j=d}$, contains at most $c$ cells of $S^n_c$.
\end{definition}

\vspace{8pt}

Le, Li and Wang~\cite{LLW1990} construct solutions of $(n,c)$-coins problems \emph{separately} for
each $c \in [0,n]$.

\vspace{8pt}

\begin{definition}\label{def:is}
\hspace{1pt} \\ A square $S^n$ is an \emph{incremental} square $\Leftrightarrow$
\begin{itemize}
\item $|\,D^{n,k}_{i+j=s}\,| \le k + 1$, 
$\forall s \in [0,2(n-1)] \, , \, \forall k \in [0,n-1]$,
\item $|\,D^{n,k}_{i-j=d}\,| \le k + 1$, 
$\forall d \in [-(n-1),n-1] \, , \, \forall k \in [0,n-1]$.
\end{itemize}
\end{definition}

\vspace{8pt}

\begin{definition}\label{def:ls}
\hspace{1pt} \\ A square $S^n$ is a \emph{Latin} square $\Leftrightarrow$ 
\begin{itemize}
\item $r^n_i$ is a permutation of $[0,n-1]$, $\, \forall i \in [0,n-1] \;$,
\item $c^n_j$ is a permutation of $[0,n-1]$, $\, \forall j \in [0,n-1] \;$.
\end{itemize}
\end{definition}

\vspace{8pt}

For a Latin square, we write $L^n$ for $S^n$ and $l^n_{i,j}$ for $s^n_{i,j}$.

\vspace{8pt}

\begin{definition}\label{def:cs}
\hspace{1pt} \\ A square $S^n$ is a \emph{cyclic} square $\Leftrightarrow$
$s^n_{i + 1, j} = s^n_{i,j} + 1 \bmod n, \; \forall i,j \in [0,n-1]$.
\end{definition}

\vspace{12pt}

In what follows, $p$ is a permutation of $[0,n-1]$.

\vspace{8pt}

We write $S^{n,p}$ for the square with cell values 
$s^{n,p}_{i,j} = (i + p(j)) \bmod n, \; \forall i,j \in [0,n-1]$.

\vspace{8pt}

Clearly, $S^{n,p}$ is a cyclic Latin square, so we use notation $L^{n,p}$ and $l^{n,p}_{i,j}$.

\section[Incremental Latin Squares and (n,c)-Coins Problems]{Incremental Latin Squares and $(n,c)$-Coins Problems}

\begin{theorem}\label{thm:main1}
\hspace{1pt} \\ $\forall n \in \mathbb{N}, \; n \ge 4$, an incremental Latin $n \!\times\! n$ square
$L^n$ represents $n + 1$ solutions of $(n,c)$-coins problems \emph{simultaneously} for $c \in [0,n]$
as $S^n_c = \{L^n_{i,j} \mid l^n_{i,j} < c\}$ (notice that $S^n_0 = \emptyset$).
\end{theorem}

\begin{proof}

\vspace{8pt}

\hspace{1pt} \\ For $c = 0$, $S^n_0 = \emptyset$ has $|\,S^n_0\,| = 0 = n\,.\,0$, and every row,
column, sum-diagonal, and diff-diagonal contains $0 \le 0$ cells of $S^n_0$.

\vspace{8pt}

For $c \in [1,n]$, we have $c - 1 \in [0,n-1]$ and
$S^n_c = \{L^n_{i,j} \mid l^n_{i,j} \le c - 1\}$. By Definition~\ref{def:ls} and
Definition~\ref{def:is}:
\begin{itemize}
\item Since each row $r^n_i$ is a permutation of $[0,n-1]$, restricting $j \mapsto l^n_{i,j}$ to
$\{j \in [0,n-1] \mid l^n_{i,j} \le c - 1\}$ yields a bijection onto $[0,c-1]$, so
$|\,S^n_c \cap R^n_i\,| = |\,[0,c-1]\,| = c \le c, \; \forall i \in [0,n-1]$.
\item Summing over all $n$ rows gives
$|\,S^n_c\,| = \sum_{i=0}^{n-1} |\,S^n_c \cap R^n_i\,| = n\,.\,c$.
\item Since each column $c^n_j$ is a permutation of $[0,n-1]$, restricting $i \mapsto l^n_{i,j}$ to
$\{i \in [0,n-1] \mid l^n_{i,j} \le c - 1\}$ yields a bijection onto $[0,c-1]$, so
$|\,S^n_c \cap C^n_j\,| = |\,[0,c-1]\,| = c \le c, \; \forall j \in [0,n-1]$.
\item For every sum-diagonal $D^n_{i+j=s}$, $S^n_c \cap D^n_{i+j=s} = D^{n,c-1}_{i+j=s}$, and since
$L^n$ is incremental, $|\,D^{n,c-1}_{i+j=s}\,| \le (c - 1) + 1 = c, \; \forall s \in [0,2(n-1)]$.
\item For every diff-diagonal $D^n_{i-j=d}$, $S^n_c \cap D^n_{i-j=d} = D^{n,c-1}_{i-j=d}$, and since
$L^n$ is incremental, $|\,D^{n,c-1}_{i-j=d}\,| \le (c - 1) + 1 = c, \; \forall d \in [-(n-1),n-1]$.
\end{itemize}
Therefore $S^n_c$ is a solution of the $(n,c)$-coins problem (Definition~\ref{def:nc}) for every
$c \in [0,n]$.

\end{proof}

\section{Injective Majorization}

Let $L^{n,p}$ be a cyclic Latin square as above and let $D$ be any diagonal.

\vspace{8pt}

Instead of using cardinality counting, which would computationally explode for large $n$, the
\emph{injective majorization principle} reduces it, for \emph{all} thresholds $k \in [0,n-1]$,
simultaneously to construct, for each diagonal direction ($D = D^n_{i+j=s}$ or $D = D^n_{i-j=d}$),
to a single majorized injection, for \emph{all} diagonals involved.

\vspace{8pt} 

\begin{lemma}\label{lem:inj-maj}
\hspace{1pt} \\ If there exists a function $f : [0,n-1]^2 \to \mathbb{N}$ such that:

\vspace{8pt}

\begin{enumerate}[nosep,label=(\arabic*)]
  \item Majorization: $\forall i,j \in [0,n-1], f(i,j) \le l^{n,p}_{i,j}$ on $D$.
  \item Injectivity: $f$ is injective on $D$.
\end{enumerate}

\vspace{8pt}

Then for every threshold $k \in [0,n-1]$, and every diagonal
$D^k$ ($D^{n,k}_{i+j=s}$ or $D^{n,k}_{i-j=d}$), $|\,D^k\,| \le k + 1$.
\end{lemma}

\begin{proof}

\vspace{8pt}

\hspace{1pt} \\ By condition~(1), $0 \le f(i,j) \le l^{n,p}_{i,j} \le k$ on $D^k$,
therefore $f(i,j) \in [0,k]$.
\\ By condition~(2), the restriction of $f$ to $D^k$ is an injective map into $[0,k]$,
therefore $|D^k| \le |\,[0,k]\,| = k + 1$.

\end{proof}

\section{Incremental Cyclic Latin Squares}

We construct an incremental cyclic Latin square $L^{n,p}$ for every $n \ge 4$ with $n \notin \{4,6\}$
using families, Family $f$, for $f \in \{1, 2, 3, 4\}$, corresponding to the subsets of the
abstract, together with the four small cases $n \in \{8, 9, 15, 21\}$. We define $m := n/2$ for
families $f = 2$ and $f = 4$, and $k := n/3$ for families $f = 3$ and $f = 4$. Where used, $m$ and
$k$ are integers according to the corresponding subsets.

\vspace{8pt}

In what follows, for any integer $x \in \mathbb{Z}$, $x \bmod n \in [0,n-1]$ denotes the unique
remainder of dividing $x$ by $n$. Throughout the proofs below, every inequality between modular
expressions is established in two steps:
\begin{enumerate}[nosep,label=(\roman*)]
  \item \emph{Modular equational reduction}: using $(x + qn) \bmod n = x \bmod n$ ($q \in \mathbb{Z}$),
  together with $n = 2m$ or $n = 3k$, to simplify an expression to $E \bmod n$.
  \item \emph{Range verification}: proving $0 \le E < n$ from the index bounds and family threshold,
  so that $E \bmod n = E$ as an integer in $[0,n-1]$.
\end{enumerate}

\vspace{8pt}

For a permutation $p$ of $[0,n-1]$, we introduce two step functions on $[0,n-1]$

\[
    j \mapsto s^{sum}_{p,j} := (p(j) - j) \bmod n
    \;\;\text{and}\;\;
    j \mapsto s^{dif\!f}_{p,j} := (p(j) + j) \bmod n.
\]

\vspace{8pt}

The step functions can be used to deal with diagonal collisions since
\[
    l^{n,p}_{s-j,j} = (s + s^{sum}_{p,j}) \bmod n
    \;\;\text{and}\;\;
    l^{n,p}_{d+j,j} = (d + s^{dif\!f}_{p,j}) \bmod n,
\]
so that, for fixed $s$, $j \mapsto l^{n,p}_{s-j,j}$ is injective $\Leftrightarrow$
$j \mapsto s^{sum}_{p,j}$ is injective on $\{j \in [0,n-1] \mid s-j \in [0,n-1]\}$, and, similarly,
for fixed $d$, $j \mapsto l^{n,p}_{d+j,j}$ is injective $\Leftrightarrow$
$j \mapsto s^{dif\!f}_{p,j}$ is injective on $\{j \in [0,n-1] \mid d+j \in [0,n-1]\}$.

\vspace{8pt}

For each family below we define $p_f$, $f^{sum}_f$, $f^{diff}_f$ and verify, for each
$D = D^n_{i+j=s}$ and $D = D^n_{i-j=d}$, separately, the two conditions of Lemma
\ref{lem:inj-maj} for the appropriate majorant. By two applications of Lemma \ref{lem:inj-maj}
(one per diagonal direction), $L^{n,p_f}$ is incremental.

\subsection[Family 1]{Family $F_1$\,: $n = 1,5 \bmod 6$, $n \ge 5$}

\paragraph{Construction of $p_1$}

\[
    p_1(j) :=
    \begin{cases}
        0,        & j = 0, \\
        n - 2j,   & 0 < 2j < n, \\
        2n - 2j,  & 2j \ge n,
    \end{cases}
    \qquad \forall j \in [0,n-1].
\]

\paragraph{Step functions}

\hspace {1pt} \\

For every $j \in [0,n-1]$, we have $p_1(j) = (-2j) \bmod n$:
\begin{itemize}[nosep]
  \item For $j = 0$: $(-2 \cdot 0) \bmod n = 0 \bmod n = 0 = p_1(0)$.
  \item For $0 < 2j < n$: $(-2j) \bmod n = (n - 2j) \bmod n = n - 2j = p_1(j)$ since $0 < n - 2j < n$.
  \item For $n \le 2j \le 2(n-1)$: $(-2j) \bmod n = (2n - 2j) \bmod n = 2n - 2j = p_1(j)$ since $n$
  is odd ($2j \ne n$) and $0 < 2 \le 2n - 2j < n$.
\end{itemize}
Since $\gcd(2,n) = 1$, $p_1$ is a permutation of $[0,n-1]$, and
\[
    s^{sum}_{p_1,j} = (p_1(j) - j) \bmod n = (-2j - j) \bmod n = (-3j) \bmod n,
\]
\[
    s^{dif\!f}_{p_1,j} = (p_1(j) + j) \bmod n = (-2j + j) \bmod n = (-j) \bmod n.
\]

Both
$j \mapsto s^{sum}_{p_1,j}$ and $j \mapsto s^{dif\!f}_{p_1,j}$ are bijections of
$[0,n-1]$, since $\gcd(3, n) = 1$ and $\gcd(1, n) = 1$, hence injective on any subset.
Therefore $j \mapsto l^{n,p_1}_{s-j,j}$ and $j \mapsto l^{n,p_1}_{d+j,j}$ are injective on
each sum-diagonal and diff-diagonal.

\paragraph{Majorants}

\hspace {1pt} \\

Since no collision can occur on any diagonal, we define
\[
    f^{sum}_1(i,j) := l^{n,p_1}_{i,j}, 
    \qquad
    f^{diff}_1(i,j) := l^{n,p_1}_{i,j}, 
    \qquad \forall i,j \in [0,n-1].
\]

\vspace{8pt}

\begin{lemma}\label{lem:fam1}
\hspace{1pt} \\ $L^{n,p_1}$ is an incremental cyclic Latin square.
\end{lemma}

\begin{proof}

\hspace{1pt} \\

\emph{Sum-diagonal direction, condition (1) of Lemma \ref{lem:inj-maj}.} 
\hspace{1pt} \\
Holds by equality.

\vspace {8pt}

\emph{Sum-diagonal direction, condition (2) of Lemma \ref{lem:inj-maj}.}
\hspace{1pt} \\
For fixed $s$, the map $j \mapsto f^{sum}_1(s-j, j) = l^{n,p_1}_{s-j,j}$ is
injective. 

\vspace{8pt}

\emph{Diff-diagonal direction, condition (1) of Lemma \ref{lem:inj-maj}.} 
\hspace{1pt} \\
Holds by equality.

\vspace{8pt}

\emph{Diff-diagonal direction, condition (2) of Lemma \ref{lem:inj-maj}.}
\hspace {1pt} \\
For fixed $d$, the map $j \mapsto f^{diff}_1(d+j, j) = l^{n,p_1}_{d+j,j}$ is
injective. 

\end{proof}

\subsection[Family 2]{Family $F_2$\,: $n = 2,4 \bmod 6$, $n \ge 10$}

$m := n/2$ is an integer with $m \ge 5$ and $2m = n$.

\paragraph{Construction of $p_2$}

\[
    p_2(j) :=
    \begin{cases}
        (2j + m + 1) \bmod n, & j \in [0,m-1], \\
        (2(j-m) + m + 4) \bmod n, & j \in [m, n-1].
    \end{cases}
\]

Since $2m = n$, the first block $j \in [0,m-1]$ steps by $+2$ through all $m$ residues of parity
$(m+1) \bmod 2$, and the second block $j \in [m,n-1]$ steps by $+2$ through all $m$ residues of
 the opposite parity $(m+4) \bmod 2$, so $p_2$ is a permutation of $[0,n-1]$.

\paragraph{Step functions}

\hspace {1pt} \\

For $j \in [0,m-1]$:
\[
    s^{sum}_{p_2,j} = (p_2(j) - j) \bmod n = (2j + m + 1 - j) \bmod n = (j + m + 1) \bmod n,
\]
\[
    s^{dif\!f}_{p_2,j} = (p_2(j) + j) \bmod n = (2j + m + 1 + j) \bmod n = (3j + m + 1) \bmod n.
\]
For $j \in [m,n-1]$:
\[
    s^{sum}_{p_2,j} = (p_2(j) - j) \bmod n = (2(j-m) + m + 4 - j) \bmod n = (j - m + 4) \bmod n,
\]
\[
    s^{dif\!f}_{p_2,j} = (p_2(j) + j) \bmod n = (2(j-m) + m + 4 + j) \bmod n = (3j - m + 4) \bmod n.
\]

\begin{center}
\begin{tabular}{@{}l c c c@{}}
\toprule
Block & $s^{sum}_{p_2,j}$ & $s^{dif\!f}_{p_2,j}$ & step in $j$ \\
\midrule
$[0,m-1]$ & $(j + m + 1) \bmod n$ & $(3j + m + 1) \bmod n$ & $+1$ (sum), $+3$ (diff) \\
$[m, n-1]$ & $(j - m + 4) \bmod n$ & $(3j - m + 4) \bmod n$ & $+1$ (sum), $+3$ (diff) \\
\bottomrule
\end{tabular}
\end{center}

Both step functions are injective within each block, since $\gcd(3, n) = \gcd(1, n) = 1$.

\paragraph{Sum collisions} 

\hspace {1pt} \\

For $j_1 \in [0,m-1]$ and $j_2 \in [m,n-1]$, from $s^{sum}_{p_2,j_1} = s^{sum}_{p_2,j_2}$ we obtain:
\begin{align*}
    (j_1 + m + 1) \bmod n = (j_2 - m + 4) \bmod n
    &\iff (j_2 - j_1 - 2m + 3) \bmod n = 0 \\
    &\iff (j_2 - j_1 - n + 3) \bmod n = 0 \qquad (\text{since } 2m = n) \\
    &\iff (j_2 - j_1 + 3) \bmod n = 0.
\end{align*}
Since $j_1 \in [0,m-1]$ and $j_2 \in [m,n-1]$, we have $1 \le j_2 - j_1 \le n - 1$, hence
$4 \le j_2 - j_1 + 3 \le n + 2$. Since $n \ge 10$, the only multiple of $n$ in $[4, n+2]$ is $n$, so
$j_2 - j_1 + 3 = n$, i.e.\ $j_2 - j_1 = n - 3$. Since $j_1 \ge 0$ and $j_2 \le n-1$, this forces
$j_1 \in \{0,1,2\}$ and $j_2 = j_1 + n - 3 \in \{n-3, n-2, n-1\}$. On a common sum-diagonal
$i_1 + j_1 = i_2 + j_2$, we have $i_1 - i_2 = j_2 - j_1 = n - 3$, which similarly forces
$i_1 \in \{n-3,n-2,n-1\}$ and $i_2 \in \{0,1,2\}$.
The collision region is the pair of $3 \times 3$ corners
\[
    C^{sum}_2 := (\,\{n-3,n-2,n-1\} \times \{0,1,2\} \, , \, \{0,1,2\} \times \{n-3,n-2,n-1\}\,).
\]

\paragraph{Majorant $f^{sum}_2$} 

\hspace {1pt} \\

We shift the top-right $3 \times 3$ corner of the square:

\[
    f^{sum}_2(i,j) :=
    \begin{cases}
        j - (n-3), & i < 3 \text{ and } j \ge n-3, \\
        l^{n,p_2}_{i,j}, & \text{otherwise}.
    \end{cases}
\]

\paragraph{Diff collision} 

\hspace {1pt} \\

For $j_1 \in [0,m-1]$ and $j_2 \in [m,n-1]$, from $s^{dif\!f}_{p_2,j_1} = s^{dif\!f}_{p_2,j_2}$
we obtain:
\begin{align*}
    (3j_1 + m + 1) \bmod n = (3j_2 - m + 4) \bmod n
    &\iff (3(j_2 - j_1) - 2m + 3) \bmod n = 0 \\
    &\iff (3(j_2 - j_1) - n + 3) \bmod n = 0 \qquad (\text{since } 2m = n) \\
    &\iff 3(j_2 - j_1 + 1) \bmod n = 0 \\
    &\iff (j_2 - j_1 + 1) \bmod n = 0 \qquad (\text{since } \gcd(3,n) = 1).
\end{align*}
Since $1 \le j_2 - j_1 \le n - 1$, we have $2 \le j_2 - j_1 + 1 \le n$, forcing
$j_2 - j_1 + 1 = n$, i.e.\ $j_2 - j_1 = n - 1$. This forces $j_1 = 0$ and $j_2 = n - 1$. On a
common diff-diagonal $i_2 - i_1 = j_2 - j_1 = n - 1$, this forces $i_1 = 0$ and $i_2 = n - 1$,
giving the single pair $\{(0,0), (n-1,n-1)\}$ (on the main diagonal $d = 0$).

\paragraph{Majorant $f^{diff}_2$} 

\hspace {1pt} \\

We shift the bottom-right corner cell:

\[
    f^{diff}_2(i,j) :=
    \begin{cases}
        1, & (i,j) = (n-1,n-1), \\
        l^{n,p_2}_{i,j}, & \text{otherwise}.
    \end{cases}
\]

\begin{lemma}\label{lem:fam2}
\hspace{1pt} \\ $L^{n,p_2}$ is an incremental cyclic Latin square.
\end{lemma}

\begin{proof}

\hspace{1pt} \\

\emph{Sum-diagonal direction, condition (1) of Lemma \ref{lem:inj-maj}.} 
\hspace{1pt} \\
On the shifted $3 \times 3$ corner ($i \in [0,2]$ and $j = n - 3 + l$ with $l \in [0,2]$), we have
$j \ge n - 3 = 2m - 3 \ge m$ (since $m \ge 5$), so:
\begin{align*}
    l^{n,p_2}_{i,n-3+l}
    &= (i + p_2(n - 3 + l)) \bmod n \\
    &= (i + (2(n - 3 + l - m) + m + 4)) \bmod n \\
    &= (i + 2n - 6 + 2l - 2m + m + 4) \bmod n \\
    &= (i + 2n - 6 + 2l - n + m + 4) \bmod n \qquad (\text{since } 2m = n) \\
    &= (i + n - 6 + 2l + m + 4) \bmod n \\
    &= (i + n + m - 2 + 2l) \bmod n \\
    &= (i + m - 2 + 2l) \bmod n.
\end{align*}
Since $i \in [0,2]$, $l \in [0,2]$, and $m \ge 5$, we have
\[
    0 < 3 \le m - 2 \le i + m - 2 + 2l \le 2 + m - 2 + 4 = m + 4 < 2m = n,
\]
so $(i + m - 2 + 2l) \bmod n = i + m - 2 + 2l$. Meanwhile, $f^{sum}_2(i,n-3+l) = l \in [0,2]$, hence
\[
    0 \le f^{sum}_2(i,n-3+l) = l \le 2 < 3 \le m - 2 \le i + m - 2 + 2l = l^{n,p_2}_{i,n-3+l}.
\]
Off the corner, $f^{sum}_2(i,j) = l^{n,p_2}_{i,j}$ holds by equality.

\vspace {8pt}

\emph{Sum-diagonal direction, condition (2) of Lemma \ref{lem:inj-maj}.}
\hspace{1pt} \\
Fix a sum-diagonal $s$. Between two shifted corner cells $(s-j_2, j_2)$ on $s$,
$f^{sum}_2(s-j_2, j_2) = j_2 - (n-3)$ is injective in $j_2$. Between two unshifted cells on $s$,
$f^{sum}_2 = l^{n,p_2}$ is injective because the only cross-block sum collisions lie in $C^{sum}_2$,
whose top-right corner is shifted. Finally, consider an unshifted cell $(i_1, j_1)$ and a shifted
corner cell $(i_2, j_2)$ ($i_2 \in [0,2]$, $j_2 \in [n-3,n-1]$) on the same sum-diagonal
$i_1 + j_1 = i_2 + j_2 = s$, where $f^{sum}_2(i_2, j_2) = j_2 - (n-3) = (j_2 - n + 3) \bmod n$:
\begin{itemize}[nosep]
  \item If $j_1 \in [0,m-1]$, using $i_1 = i_2 + j_2 - j_1$ and $2m = n$:
  \begin{align*}
      (l^{n,p_2}_{i_1,j_1} - f^{sum}_2(i_2,j_2)) \bmod n
      &= ((i_1 + 2j_1 + m + 1) - (j_2 - n + 3)) \bmod n \\
      &= (i_2 + j_2 + j_1 + m + 1 - j_2 + 2m - 3) \bmod n \\
      &= (i_2 + j_1 + n + m - 2) \bmod n \\
      &= (i_2 + j_1 + m - 2) \bmod n.
  \end{align*}
  Since $i_2 \in [0,2]$, $j_1 \in [0,m-1]$, and $m \ge 5$, we have
  $0 < 3 \le m - 2 \le i_2 + j_1 + m - 2 \le 2 + (m-1) + m - 2 = 2m - 1 = n - 1 < n$,
  so $(l^{n,p_2}_{i_1,j_1} - f^{sum}_2(i_2,j_2)) \bmod n = i_2 + j_1 + m - 2 \ne 0$.
  \item If $j_1 \in [m,n-1]$ (unshifted), using $i_1 = i_2 + j_2 - j_1$:
  \begin{align*}
      (l^{n,p_2}_{i_1,j_1} - f^{sum}_2(i_2,j_2)) \bmod n
      &= ((i_1 + 2(j_1 - m) + m + 4) - (j_2 - n + 3)) \bmod n \\
      &= (i_2 + j_2 + j_1 - m + 4 - j_2 + n - 3) \bmod n \\
      &= (i_2 + (j_1 - m) + n + 1) \bmod n \\
      &= (i_2 + (j_1 - m) + 1) \bmod n.
  \end{align*}
  Since $i_2 \in [0,2]$, $j_1 - m \in [0,m-1]$, and $m \ge 5$, we have
  $0 < 1 \le i_2 + (j_1 - m) + 1 \le 2 + (m-1) + 1 = m + 2 < 2m = n$,
  so $(l^{n,p_2}_{i_1,j_1} - f^{sum}_2(i_2,j_2)) \bmod n = i_2 + (j_1 - m) + 1 \ne 0$.
\end{itemize}
Thus no unshifted cell can collide with a shifted corner cell on $s$.

\vspace {8pt}

\emph{Diff-diagonal direction, condition (1) of Lemma \ref{lem:inj-maj}.} 
\hspace{1pt} \\
On the single shifted cell $(n-1,n-1)$:
\begin{align*}
    l^{n,p_2}_{n-1,n-1}
    &= (n - 1 + p_2(n - 1)) \bmod n \\
    &= (n - 1 + (2(n - 1 - m) + m + 4)) \bmod n \\
    &= (n - 1 + 2n - 2 - 2m + m + 4) \bmod n \\
    &= (n - 1 + 2n - 2 - n + m + 4) \bmod n \qquad (\text{since } 2m = n) \\
    &= (2n + m + 1) \bmod n \\
    &= (m + 1) \bmod n.
\end{align*}
Since $m \ge 5$, we have $0 < 6 \le m + 1 < 2m = n$, so $(m + 1) \bmod n = m + 1$, while
$f^{diff}_2(n-1,n-1) = 1$, giving
\[
    0 < f^{diff}_2(n-1,n-1) = 1 < 6 \le m + 1 = l^{n,p_2}_{n-1,n-1}.
\]
Off $(n-1,n-1)$, $f^{diff}_2(i,j) = l^{n,p_2}_{i,j}$ holds by equality.

\vspace {8pt}

\emph{Diff-diagonal direction, condition (2) of Lemma \ref{lem:inj-maj}.} 
\hspace{1pt} \\
For $d \ne 0$, $f^{diff}_2 = l^{n,p_2}$ on $D^n_{i-j=d}$ and no diff collision occurs. On the main
diagonal $d = 0$ ($i = j$), the unshifted cells $(j,j)$ for $j \in [0,n-2]$ have pairwise distinct
$l^{n,p_2}$-values by the diff-collision analysis. Moreover, none of them equals
$f^{diff}_2(n-1,n-1) = 1$:
\begin{itemize}[nosep]
  \item For $j \in [0,m-1]$:
  \begin{align*}
      l^{n,p_2}_{j,j} = 1
      &\iff (3j + m + 1) \bmod n = 1 \\
      &\iff (3j + m) \bmod n = 0 \\
      &\iff (3j + m - 2n) \bmod n = 0 \\
      &\iff 3(j - m) \bmod n = 0 \qquad (\text{since } 2n = 4m) \\
      &\iff (j - m) \bmod n = 0 \qquad (\text{since } \gcd(3,n) = 1),
  \end{align*}
  which requires $j = m \notin [0,m-1]$.
  \item For $j \in [m,n-2]$:
  \begin{align*}
      l^{n,p_2}_{j,j} = 1
      &\iff (3j - m + 4) \bmod n = 1 \\
      &\iff (3j - m + 3) \bmod n = 0 \\
      &\iff (3j - m + 3 - n) \bmod n = 0 \\
      &\iff 3(j + 1 - m) \bmod n = 0 \qquad (\text{since } n = 2m) \\
      &\iff (j + 1 - m) \bmod n = 0 \qquad (\text{since } \gcd(3,n) = 1),
  \end{align*}
  which requires $j = m - 1 \notin [m,n-2]$.
\end{itemize}

\end{proof}

\subsection[Family 3]{Family $F_3$\,: $n = 3 \bmod 6$, $n \ge 27$}

$k := n/3$ is an integer with $k \ge 9$ and $3k = n$.

\paragraph{Construction of $p_3$}

\[
    p_3(j) :=
    \begin{cases}
        (3j + 2k) \bmod n,          & j \in [0,k-1], \\
        (3(j-k) + 2k + 4) \bmod n,  & j \in [k, 2k-1], \\
        (3(j-2k) + 2k + 8) \bmod n, & j \in [2k, n-1].
    \end{cases}
\]

Since $3k = n$ and $\gcd(4,3) = 1$, the three blocks $I_u := [uk, (u+1)k-1]$ for $u \in \{0,1,2\}$
step by $+3$ through the three distinct residue classes $(2k + 4u) \bmod 3$, so $p_3$ is a
permutation of $[0,n-1]$.

\paragraph{Step functions}

\hspace {1pt} \\

Since $3k = n$, on each block $I_u = [uk, (u+1)k-1]$ ($u \in \{0,1,2\}$) we have
\[
    p_3(j) = (3(j - uk) + 2k + 4u) \bmod n = (3j - un + 2k + 4u) \bmod n = (3j + 2k + 4u) \bmod n,
\]
and therefore
\[
    s^{sum}_{p_3,j} = (p_3(j) - j) \bmod n = (2j + 2k + 4u) \bmod n,
    \qquad
    s^{dif\!f}_{p_3,j} = (p_3(j) + j) \bmod n = (4j + 2k + 4u) \bmod n.
\]

\begin{center}
\begin{tabular}{@{}l c c c@{}}
\toprule
Block & $s^{sum}_{p_3,j}$ & $s^{dif\!f}_{p_3,j}$ & step in $j$ \\
\midrule
$[0,k-1]$ ($u=0$)   & $(2j + 2k) \bmod n$     & $(4j + 2k) \bmod n$     & $+2$ (sum), $+4$ (diff) \\
$[k, 2k-1]$ ($u=1$) & $(2j + 2k + 4) \bmod n$ & $(4j + 2k + 4) \bmod n$ & $+2$ (sum), $+4$ (diff) \\
$[2k, n-1]$ ($u=2$) & $(2j + 2k + 8) \bmod n$ & $(4j + 2k + 8) \bmod n$ & $+2$ (sum), $+4$ (diff) \\
\bottomrule
\end{tabular}
\end{center}

Both step functions are injective within each block, since $\gcd(2,n) = \gcd(4,n) = 1$.

\paragraph{Sum collisions} 

\hspace {1pt} \\

For $j_1 \in I_{u_1}$ and $j_2 \in I_{u_2}$ with $0 \le u_1 < u_2 \le 2$, from
$s^{sum}_{p_3,j_1} = s^{sum}_{p_3,j_2}$ we obtain:
\begin{align*}
    &(2j_1 + 2k + 4u_1) \bmod n = (2j_2 + 2k + 4u_2) \bmod n \\
    &\qquad\iff 2(j_2 - j_1 + 2(u_2 - u_1)) \bmod n = 0 \\
    &\qquad\iff (j_2 - j_1 + 2(u_2 - u_1)) \bmod n = 0 \qquad (\text{since } \gcd(2,n) = 1).
\end{align*}
\begin{itemize}[nosep]
  \item For adjacent blocks ($u_2 - u_1 = 1$, i.e.\ $[0,k-1] \times [k,2k-1]$ or
  $[k,2k-1] \times [2k,n-1]$), we have $1 \le j_2 - j_1 \le 2k - 1$, so
  $0 < 3 \le j_2 - j_1 + 2 \le 2k + 1 < 3k = n$ (as $k \ge 9$). Hence
  $(j_2 - j_1 + 2) \bmod n = j_2 - j_1 + 2 \ne 0$, and no collision occurs.
  \item For outer blocks ($u_1 = 0, u_2 = 2$, i.e.\ $j_1 \in [0,k-1]$ and $j_2 \in [2k,n-1]$), we
  have $(j_2 - j_1 + 4) \bmod n = 0$ with $k + 1 \le j_2 - j_1 \le n - 1$, so
  $k + 5 \le j_2 - j_1 + 4 \le n + 3$. The only multiple of $n$ in $[k+5, n+3]$ is $n$, forcing
  $j_2 - j_1 + 4 = n$, i.e.\ $j_2 - j_1 = n - 4$. Since $j_1 \ge 0$ and $j_2 \le n - 1$, this gives
  $j_1 \in \{0,1,2,3\}$ and $j_2 = j_1 + n - 4 \in \{n-4, n-3, n-2, n-1\}$. On a common
  sum-diagonal $i_1 + j_1 = i_2 + j_2$, $i_1 - i_2 = j_2 - j_1 = n - 4$ forces
  $i_1 \in \{n-4,\dots,n-1\}$ and $i_2 \in \{0,1,2,3\}$.
\end{itemize}
The collision region is the pair of $4 \times 4$ corners
\[
    C^{sum}_3 := (\, \{n-4,\dots,n-1\} \times \{0,\dots,3\}
    \, , \, \{0,\dots,3\} \times \{n-4,\dots,n-1\} \,).
\]

\paragraph{Majorant $f^{sum}_3$} 

\hspace {1pt} \\

We shift the top-right $4 \times 4$ corner of the square in a column-dependent amount:

\[
    f^{sum}_3(i,j) :=
    \begin{cases}
        l^{n,p_3}_{i,j} - k,  & i < 4 \text{ and } j \in \{n-4, n-3\}, \\
        l^{n,p_3}_{i,j} - 2k, & i < 4 \text{ and } j \in \{n-2, n-1\}, \\
        l^{n,p_3}_{i,j},      & \text{otherwise}.
    \end{cases}
\]

\paragraph{Diff collision} 

\hspace {1pt} \\

For $j_1 \in I_{u_1}$ and $j_2 \in I_{u_2}$ with $0 \le u_1 < u_2 \le 2$, from
$s^{dif\!f}_{p_3,j_1} = s^{dif\!f}_{p_3,j_2}$ we obtain:
\begin{align*}
    &(4j_1 + 2k + 4u_1) \bmod n = (4j_2 + 2k + 4u_2) \bmod n \\
    &\qquad\iff 4(j_2 - j_1 + (u_2 - u_1)) \bmod n = 0 \\
    &\qquad\iff (j_2 - j_1 + (u_2 - u_1)) \bmod n = 0 \qquad (\text{since } \gcd(4,n) = 1).
\end{align*}
\begin{itemize}[nosep]
  \item For adjacent blocks ($u_2 - u_1 = 1$), $1 \le j_2 - j_1 \le 2k - 1$ gives
  $0 < 2 \le j_2 - j_1 + 1 \le 2k < 3k = n$, so $(j_2 - j_1 + 1) \bmod n = j_2 - j_1 + 1 \ne 0$
  and no collision occurs.
  \item For outer blocks ($u_1 = 0, u_2 = 2$), $(j_2 - j_1 + 2) \bmod n = 0$ with
  $k + 3 \le j_2 - j_1 + 2 \le n + 1$ forces $j_2 - j_1 + 2 = n$, i.e.\ $j_2 - j_1 = n - 2$, giving
  $j_1 \in \{0,1\}$ and $j_2 \in \{n-2, n-1\}$. On a common diff-diagonal
  $i_2 - i_1 = j_2 - j_1 = n - 2$, this forces $i_1 \in \{0,1\}$ and $i_2 \in \{n-2, n-1\}$.
\end{itemize}
Therefore every diff collision lies in the pair of $2 \times 2$ corners
$(\{0,1\}^2 , \{n-2,n-1\}^2)$.

\paragraph{Majorant $f^{diff}_3$} 

\hspace {1pt} \\

We shift the bottom-right $2 \times 2$ corner of the square in a column-dependent amount:

\[
    f^{diff}_3(i,j) :=
    \begin{cases}
        l^{n,p_3}_{i,j} - 2k, & i \ge n-2 \text{ and } j = n-2, \\
        l^{n,p_3}_{i,j} - k,  & i \ge n-2 \text{ and } j = n-1, \\
        l^{n,p_3}_{i,j},      & \text{otherwise}.
    \end{cases}
\]

\begin{lemma}\label{lem:fam3}
\hspace{1pt} \\ $L^{n,p_3}$ is an incremental cyclic Latin square.
\end{lemma}

\begin{proof}

\hspace{1pt} \\

\emph{Sum-diagonal direction, condition (1) of Lemma \ref{lem:inj-maj}.} 
\hspace{1pt} \\
On the shifted top-right $4 \times 4$ corner ($i \in [0,3]$ and $j = n - 4 + l$ with
$l \in [0,3]$), we have $j \ge n - 4 = 3k - 4 \ge 2k$ (since $k \ge 9$), so:
\begin{align*}
    l^{n,p_3}_{i,n-4+l}
    &= (i + p_3(n - 4 + l)) \bmod n \\
    &= (i + (3(n - 4 + l - 2k) + 2k + 8)) \bmod n \\
    &= (i + 3n - 12 + 3l - 6k + 2k + 8) \bmod n \\
    &= (i + 3n - 12 + 3l - 2n + 2k + 8) \bmod n \qquad (\text{since } 3k = n) \\
    &= (i + n + 2k - 4 + 3l) \bmod n \\
    &= (i + 2k - 4 + 3l) \bmod n.
\end{align*}
Since $i \in [0,3]$, $l \in [0,3]$, and $k \ge 9$, we have
\[
    0 < 14 \le 2k - 4 \le i + 2k - 4 + 3l \le 3 + 2k - 4 + 9 = 2k + 8 < 3k = n,
\]
so $(i + 2k - 4 + 3l) \bmod n = i + 2k - 4 + 3l$. Consequently:
\begin{itemize}[nosep]
  \item For $l \in \{0,1\}$ ($j \in \{n-4, n-3\}$), $l^{n,p_3}_{i,n-4+l} = i + 2k - 4 + 3l \ge 2k - 4 > k$
  (as $k \ge 9$), so
  \[
      0 < k - 4 \le i + k - 4 + 3l = l^{n,p_3}_{i,j} - k = f^{sum}_3(i,j) < l^{n,p_3}_{i,j}.
  \]
  \item For $l \in \{2,3\}$ ($j \in \{n-2, n-1\}$), $l^{n,p_3}_{i,n-4+l} = i + 2k - 4 + 3l \ge 2k + 2 > 2k$,
  so
  \[
      0 < 2 \le i - 4 + 3l = l^{n,p_3}_{i,j} - 2k = f^{sum}_3(i,j) < l^{n,p_3}_{i,j}.
  \]
\end{itemize}
Off the corner, $f^{sum}_3(i,j) = l^{n,p_3}_{i,j}$ holds by equality.

\vspace{8pt}

\emph{Sum-diagonal direction, condition (2) of Lemma \ref{lem:inj-maj}.} 
\hspace{1pt} \\
Fix a sum-diagonal $s = i + j$. For any shifted corner cell $(i, n-4+l)$ with $i \in [0,3]$ and
$l \in [0,3]$, substituting $i = s - (n - 4 + l) = s - 3k + 4 - l$ into the formula from
condition~(1) gives:
\begin{itemize}[nosep]
  \item For $l \in \{0,1\}$ ($j \in \{n-4, n-3\}$), since $0 < i + k - 4 + 3l \le k + 2 < n$:
  \begin{align*}
      f^{sum}_3(i,n-4+l)
      &= (i + k - 4 + 3l) \bmod n \\
      &= (s - 3k + 4 - l + k - 4 + 3l) \bmod n \\
      &= (s - n + k + 2l) \bmod n \\
      &= (s + k + 2l) \bmod n.
  \end{align*}
  \item For $l \in \{2,3\}$ ($j \in \{n-2, n-1\}$), since $0 < i - 4 + 3l \le 8 < n$:
  \begin{align*}
      f^{sum}_3(i,n-4+l)
      &= (i - 4 + 3l) \bmod n \\
      &= (s - 3k + 4 - l - 4 + 3l) \bmod n \\
      &= (s - n + 2l) \bmod n \\
      &= (s + 2l) \bmod n.
  \end{align*}
\end{itemize}
Thus the four corner columns $l = 0, 1, 2, 3$ take values $(s + v_l) \bmod n$ with step offsets
\[
    v_0 = k, \qquad v_1 = k + 2, \qquad v_2 = 4, \qquad v_3 = 6.
\]
Since $k \ge 9$, the four offsets $\{4, 6, k, k+2\} \subset [0,n-1]$ are pairwise distinct, so the
shifted corner cells on $s$ are pairwise distinct. Moreover, none of these four values is taken by
any unshifted cell $(s-j_1, j_1)$ on $s$, where $l^{n,p_3}_{s-j_1,j_1} = (s + s^{sum}_{p_3,j_1}) \bmod n$
and $s^{sum}_{p_3,j_1} = (2j_1 + 2k + 4u_1) \bmod n$ for $j_1 \in I_{u_1}$ ($u_1 \in \{0,1,2\}$):
\begin{itemize}[nosep]
  \item For $v_l = k + 2l$ ($l \in \{0,1\}$), using $k = n - 2k$ and $\gcd(2,n) = 1$:
  \begin{align*}
      (2j_1 + 2k + 4u_1) \bmod n = (k + 2l) \bmod n
      &\iff (2j_1 + k + 4u_1 - 2l) \bmod n = 0 \\
      &\iff 2(j_1 - k + 2u_1 - l) \bmod n = 0 \\
      &\iff (j_1 - k + 2u_1 - l) \bmod n = 0,
  \end{align*}
  which requires $j_1 = k - 2u_1 + l$: for $u_1 = 0$ this is $k + l \notin [0,k-1]$; for $u_1 = 1$
  this is $k - 2 + l \in \{k-2,k-1\} \notin [k,2k-1]$; for $u_1 = 2$ this is
  $k - 4 + l \notin [2k,n-1]$.
  \item For $v_l = 2l$ ($l \in \{2,3\}$), using $2k = 2n - 4k$ and $\gcd(2,n) = 1$:
  \begin{align*}
      (2j_1 + 2k + 4u_1) \bmod n = 2l \bmod n
      &\iff (2j_1 - 4k + 4u_1 - 2l) \bmod n = 0 \\
      &\iff 2(j_1 - 2k + 2u_1 - l) \bmod n = 0 \\
      &\iff (j_1 - 2k + 2u_1 - l) \bmod n = 0,
  \end{align*}
  which requires $j_1 = 2k - 2u_1 + l$: for $u_1 = 0$ this is $2k + l \notin [0,k-1]$; for $u_1 = 1$
  this is $2k - 2 + l \in \{2k,2k+1\} \notin [k,2k-1]$; for $u_1 = 2$ this is
  $2k - 4 + l \in \{2k-2,2k-1\} \notin [2k,n-1]$.
\end{itemize}
Hence $f^{sum}_3$ is injective on $D^n_{i+j=s}$.

\vspace{8pt}

\emph{Diff-diagonal direction, condition (1) of Lemma \ref{lem:inj-maj}.} 
\hspace{1pt} \\
On the shifted bottom-right $2 \times 2$ corner ($i = n - 2 + r$ with $r \in \{0,1\}$ and
$j = n - 2 + l$ with $l \in \{0,1\}$):
\begin{align*}
    l^{n,p_3}_{n-2+r,n-2+l}
    &= (n - 2 + r + p_3(n - 2 + l)) \bmod n \\
    &= (n - 2 + r + (3(n - 2 + l - 2k) + 2k + 8)) \bmod n \\
    &= (n - 2 + r + 3n - 6 + 3l - 6k + 2k + 8) \bmod n \\
    &= (n - 2 + r + 3n - 6 + 3l - 2n + 2k + 8) \bmod n \qquad (\text{since } 3k = n) \\
    &= (2n + r + 2k + 3l) \bmod n \\
    &= (r + 2k + 3l) \bmod n.
\end{align*}
Since $r \in \{0,1\}$, $l \in \{0,1\}$, and $k \ge 9$, we have
\[
    0 < 18 \le 2k \le r + 2k + 3l \le 1 + 2k + 3 = 2k + 4 < 3k = n,
\]
so $(r + 2k + 3l) \bmod n = r + 2k + 3l$. Consequently:
\begin{itemize}[nosep]
  \item For $l = 0$ ($j = n - 2$): $l^{n,p_3}_{n-2+r,n-2} = r + 2k \ge 2k$, so
  \[
      0 \le r = l^{n,p_3}_{n-2+r,n-2} - 2k = f^{diff}_3(n-2+r,n-2) < l^{n,p_3}_{n-2+r,n-2}.
  \]
  \item For $l = 1$ ($j = n - 1$): $l^{n,p_3}_{n-2+r,n-1} = r + 2k + 3 > k$, so
  \[
      0 < k + 3 \le r + k + 3 = l^{n,p_3}_{n-2+r,n-1} - k = f^{diff}_3(n-2+r,n-1) < l^{n,p_3}_{n-2+r,n-1}.
  \]
\end{itemize}
Off the corner, $f^{diff}_3(i,j) = l^{n,p_3}_{i,j}$ holds by equality.

\vspace{8pt}

\emph{Diff-diagonal direction, condition (2) of Lemma \ref{lem:inj-maj}.} 
\hspace{1pt} \\
Fix a diff-diagonal $d = i - j$. For a shifted corner cell $(n-2+r, n-2+l)$ ($r, l \in \{0,1\}$),
we have $d = r - l$, i.e.\ $r = d + l$, so from condition~(1):
\begin{itemize}[nosep]
  \item For $l = 0$ ($j = n - 2$):
  $f^{diff}_3(n-2+r,n-2) = r = (d + 0) \bmod n$.
  \item For $l = 1$ ($j = n - 1$):
  $f^{diff}_3(n-2+r,n-1) = r + k + 3 = d + k + 4 = (d + k + 4) \bmod n$.
\end{itemize}
Since $0 \ne k + 4 \bmod n$, these two shifted values are distinct when $d = 0$. Moreover, neither
$(d + 0) \bmod n$ nor $(d + k + 4) \bmod n$ is taken by any unshifted cell $(d+j_1, j_1)$ on
$D^n_{i-j=d}$, where $l^{n,p_3}_{d+j_1,j_1} = (d + s^{dif\!f}_{p_3,j_1}) \bmod n$ and
$s^{dif\!f}_{p_3,j_1} = (4j_1 + 2k + 4u_1) \bmod n$ for $j_1 \in I_{u_1}$:
\begin{itemize}[nosep]
  \item Using $2k = 2n - 4k$ and $\gcd(4,n) = 1$:
  \[
      (4j_1 + 2k + 4u_1) \bmod n = 0
      \iff 4(j_1 - k + u_1) \bmod n = 0
      \iff (j_1 - k + u_1) \bmod n = 0,
  \]
  which requires $j_1 = k - u_1$: for $u_1 = 0$ this is $k \notin [0,k-1]$; for $u_1 = 1$ this is
  $k - 1 \notin [k,2k-1]$; for $u_1 = 2$ this is $k - 2 \notin [2k,n-1]$.
  \item Similarly, using $k - 4 = n - 2k - 4 = 3n - 8k - 4$ and $\gcd(4,n) = 1$:
  \begin{align*}
      (4j_1 + 2k + 4u_1) \bmod n = (k + 4) \bmod n
      &\iff (4j_1 + k - 4 + 4u_1) \bmod n = 0 \\
      &\iff 4(j_1 - 2k - 1 + u_1) \bmod n = 0 \\
      &\iff (j_1 - 2k - 1 + u_1) \bmod n = 0,
  \end{align*}
  which requires $j_1 = 2k + 1 - u_1$: for $u_1 = 0$ this is $2k + 1 \notin [0,k-1]$; for $u_1 = 1$
  this is $2k \notin [k,2k-1]$; for $u_1 = 2$ this is $2k - 1 \notin [2k,n-1]$.
\end{itemize}
Hence $f^{diff}_3$ is injective on $D^n_{i-j=d}$.

\end{proof}

\subsection[Family 4]{Family $F_4$\,: $n = 0 \bmod 6$, $n \ge 12$}

Here both $2 \mid n$ and $3 \mid n$, so both $m := n/2 \ge 6$ and $k := n/3 \ge 4$ are integers,
and $n = 2m = 3k$.

\paragraph{Construction of $p_4$}

\[
    p_4(j) :=
    \begin{cases}
        n - 2,                 & j = 0, \\
        2j + 2,                & j \in [1, m-3], \\
        0,                     & j = m-2, \\
        2(j - m + 1) + 1,      & j \in [m-1, n-2], \\
        2,                     & j = n-1.
    \end{cases}
\]

The three boundary pins $j \in \{0, m-2, n-1\}$ cover $\{0, 2, n-2\}$, the first main block
$j \in [1, m-3]$ covers the remaining even values $\{4, 6, \dots, n-4\}$, and the second main block
$j \in [m-1, n-2]$ covers all odd values $\{1, 3, \dots, n-1\}$, so $p_4$ is a permutation of
$[0,n-1]$.

\paragraph{Step functions}

\hspace {1pt} \\

Using $2m = n$, we compute $s^{sum}_{p_4,j} = (p_4(j) - j) \bmod n$ and
$s^{dif\!f}_{p_4,j} = (p_4(j) + j) \bmod n$ on each piece:
\begin{itemize}[nosep]
  \item $j = 0$: $s^{sum}_{p_4,0} = (n - 2) \bmod n$ and $s^{dif\!f}_{p_4,0} = (n - 2) \bmod n$.
  \item $j \in [1, m-3]$: $s^{sum}_{p_4,j} = (2j + 2 - j) \bmod n = (j + 2) \bmod n$ and
  $s^{dif\!f}_{p_4,j} = (3j + 2) \bmod n$.
  \item $j = m - 2$: $s^{sum}_{p_4,m-2} = (0 - (m - 2)) \bmod n = (2m - m + 2) \bmod n = (m + 2) \bmod n$
  and $s^{dif\!f}_{p_4,m-2} = (m - 2) \bmod n$.
  \item $j \in [m-1, n-2]$:
  $s^{sum}_{p_4,j} = (2(j - m + 1) + 1 - j) \bmod n = (j - 2m + 3) \bmod n = (j + 3) \bmod n$ and
  $s^{dif\!f}_{p_4,j} = (3j - 2m + 3) \bmod n = (3j + 3) \bmod n$.
  \item $j = n - 1$: $s^{sum}_{p_4,n-1} = (2 - (n - 1)) \bmod n = 3 \bmod n$ and
  $s^{dif\!f}_{p_4,n-1} = (2 + n - 1) \bmod n = 1 \bmod n$.
\end{itemize}

\begin{center}
\begin{tabular}{@{}l c c@{}}
\toprule
Piece & $s^{sum}_{p_4,j}$ & $s^{dif\!f}_{p_4,j}$ \\
\midrule
$\{0\}$             & $(n - 2) \bmod n$     & $(n - 2) \bmod n$ \\
$[1, m-3]$          & $(j + 2) \bmod n$     & $(3j + 2) \bmod n$ \\
$\{m-2\}$           & $(m + 2) \bmod n$     & $(m - 2) \bmod n$ \\
$[m-1, n-2]$        & $(j + 3) \bmod n$     & $(3j + 3) \bmod n$ \\
$\{n-1\}$           & $3 \bmod n$           & $1 \bmod n$ \\
\bottomrule
\end{tabular}
\end{center}

\paragraph{Sum collisions}

\hspace {1pt} \\

Evaluating $s^{sum}_{p_4,j} \in [0,n-1]$ across the five pieces:
\begin{itemize}[nosep]
  \item $j = 0$ gives $n - 2$;
  \item $j \in [1, m-3]$ gives $j + 2 \in [3, m-1]$;
  \item $j = m - 2$ gives $m + 2$;
  \item $j \in [m-1, n-2]$ gives $(j + 3) \bmod n$, covering $[m+2, n-1]$ (for $j \in [m-1, n-4]$)
  and $\{0, 1\}$ (for $j \in \{n-3, n-2\}$);
  \item $j = n - 1$ gives $3$.
\end{itemize}
Thus the three step values $\{2, m, m+1\}$ are completely vacant, while three step values repeat:
\begin{enumerate}[nosep,label=(\arabic*)]
  \item $s^{sum}_{p_4,j} = m + 2$ at $j_1 = m - 2$ and $j_2 = m - 1$ ($j_2 - j_1 = 1$). On a
  sum-diagonal $i_1 + j_1 = i_2 + j_2$, $i_1 = i_2 + 1 \le n - 1 \iff i_2 < n - 1$.
  \item $s^{sum}_{p_4,j} = n - 2$ at $j_1 = 0$ and $j_2 = n - 5$ ($j_2 - j_1 = n - 5$). On a
  sum-diagonal $i_1 + j_1 = i_2 + j_2$, $i_1 = i_2 + n - 5 \le n - 1 \iff i_2 < 5 \iff i_2 + j_2 < n$.
  \item $s^{sum}_{p_4,j} = 3$ at $j_1 = 1$ and $j_2 = n - 1$ ($j_2 - j_1 = n - 2$). On a
  sum-diagonal $i_1 + j_1 = i_2 + j_2$, $i_1 = i_2 + n - 2 \le n - 1 \iff i_2 < 2$.
\end{enumerate}

\paragraph{Majorant $f^{sum}_4$}

\hspace {1pt} \\

We shift the three seam columns $j \in \{m-1, n-5, n-1\}$ into the three vacant step values
$\{m+1, m, 2\}$:
\[
    f^{sum}_4(i,j) :=
    \begin{cases}
        l^{n,p_4}_{i,j} - 1,         & j = m-1 \text{ and } i < n-1, \\
        l^{n,p_4}_{i,j} - (m-2),     & j = n-5 \text{ and } i + j < n, \\
        l^{n,p_4}_{i,j} - 1,         & j = n-1 \text{ and } i < 2, \\
        l^{n,p_4}_{i,j},             & \text{otherwise}.
    \end{cases}
\]

\paragraph{Diff collisions and majorant $f^{diff}_4$}

\hspace {1pt} \\

Both main blocks $[1, m-3]$ and $[m-1, n-2]$ have diff step $+3$. Because $n = 3k$, on any
diff-diagonal $d = i - j$:
\begin{itemize}[nosep]
  \item the first main block has $l^{n,p_4}_{d+j,j} = (d + 3j + 2) \bmod n \equiv d + 2 \pmod 3$,
  \item the second main block has $l^{n,p_4}_{d+j,j} = (d + 3j + 3) \bmod n \equiv d \pmod 3$,
  \item the third residue class $d + 1 \bmod 3$ is vacant across both main blocks.
\end{itemize}
Within either main block, two columns $j_1 < j_2$ collide on $D^n_{i-j=d}$ iff
$3(j_2 - j_1) \bmod{3k} = 0 \iff j_2 - j_1 = k$ (since $j_2 - j_1 \le m - 1 < 2k$), which forces
$j_2 \ge 1 + k$ (in the first block) or $j_2 \ge m - 1 + k$ (in the second block) together with
$i_2 = i_1 + k \ge k$. Shifting these wrapped cells into the vacant residue class $d + 1 \bmod 3$
defines:
\[
    f^{diff}_4(i,j) :=
    \begin{cases}
        l^{n,p_4}_{i,j} - 1, & 1+k \le j \le m-3 \text{ and } i \ge k, \\
        l^{n,p_4}_{i,j} - 1, & j = m-1+k \text{ and } k \le i < 2k, \\
        l^{n,p_4}_{i,j} - 4, & j = m-1+k \text{ and } i \ge 2k \text{ and } n = 12, \\
        l^{n,p_4}_{i,j} - 5, & j = m-1+k \text{ and } i \ge 2k \text{ and } n \ne 12, \\
        l^{n,p_4}_{i,j} - 2, & m+k \le j \le n-2 \text{ and } i \ge k, \\
        l^{n,p_4}_{i,j},     & \text{otherwise}.
    \end{cases}
\]

\begin{lemma}\label{lem:fam4}
\hspace{1pt} \\ $L^{n,p_4}$ is an incremental cyclic Latin square.
\end{lemma}

\begin{proof}

\hspace{1pt} \\

\emph{Sum-diagonal direction, condition (1) of Lemma \ref{lem:inj-maj}.}
\hspace{1pt} \\
We compute $l^{n,p_4}_{i,j}$ on each of the three shifted seam regions:
\begin{itemize}[nosep]
  \item For $j = m - 1$ and $i \in [0,n-2]$: since $p_4(m-1) = 1$ and $0 < 1 \le i + 1 \le n - 1 < n$,
  \[
      l^{n,p_4}_{i,m-1} = (i + 1) \bmod n = i + 1,
      \quad\text{so}\quad
      0 \le i = l^{n,p_4}_{i,m-1} - 1 = f^{sum}_4(i,m-1) < l^{n,p_4}_{i,m-1}.
  \]
  \item For $j = n - 5$ and $i + j < n$ (i.e.\ $i \in [0,4]$): since $n \ge 12$ ($m \ge 6$),
  $n - 5 = 2m - 5 \in [m-1, n-2]$, so:
  \begin{align*}
      l^{n,p_4}_{i,n-5}
      &= (i + p_4(n - 5)) \bmod n \\
      &= (i + (2(n - 5 - m + 1) + 1)) \bmod n \\
      &= (i + 2n - 8 - 2m + 1) \bmod n \\
      &= (i + 2n - 8 - n + 1) \bmod n \qquad (\text{since } 2m = n) \\
      &= (i + n - 7) \bmod n.
  \end{align*}
  Since $i \in [0,4]$ and $n \ge 12$, we have $0 < 5 \le n - 7 \le i + n - 7 \le n - 3 < n$, so
  $(i + n - 7) \bmod n = i + n - 7$. Since $n - 7 = 2m - 7 \ge m - 1 > m - 2$ (as $m \ge 6$),
  \[
      0 < 1 \le m - 5 \le i + m - 5 = (i + n - 7) - (m - 2) = f^{sum}_4(i,n-5) < l^{n,p_4}_{i,n-5}.
  \]
  \item For $j = n - 1$ and $i \in \{0,1\}$: since $p_4(n-1) = 2$ and $0 < 2 \le i + 2 \le 3 < n$,
  \[
      l^{n,p_4}_{i,n-1} = (i + 2) \bmod n = i + 2,
      \quad\text{so}\quad
      0 < 1 \le i + 1 = l^{n,p_4}_{i,n-1} - 1 = f^{sum}_4(i,n-1) < l^{n,p_4}_{i,n-1}.
  \]
\end{itemize}
Off the three seams, $f^{sum}_4(i,j) = l^{n,p_4}_{i,j}$ holds by equality.

\vspace{8pt}

\emph{Sum-diagonal direction, condition (2) of Lemma \ref{lem:inj-maj}.}
\hspace{1pt} \\
Fix a sum-diagonal $s = i + j$. Substituting $i = s - j$ into the non-modular formulas from
condition~(1) on the three shifted seam regions gives:
\begin{itemize}[nosep]
  \item At $j = m - 1$ ($i < n - 1$):
  $f^{sum}_4(i,m-1) = i = (s - (m - 1)) \bmod n = (s + 2m - m + 1) \bmod n = (s + m + 1) \bmod n$.
  \item At $j = n - 5$ ($i < 5$):
  $f^{sum}_4(i,n-5) = i + m - 5 = (s - (n - 5) + m - 5) \bmod n = (s - n + m) \bmod n = (s + m) \bmod n$.
  \item At $j = n - 1$ ($i < 2$):
  $f^{sum}_4(i,n-1) = i + 1 = (s - (n - 1) + 1) \bmod n = (s - n + 2) \bmod n = (s + 2) \bmod n$.
\end{itemize}
Thus the three seam shifts replace the repeated step values $\{m+2, n-2, 3\}$ on $s$ by the three
vacant step values $\{m+1, m, 2\}$, which are pairwise distinct (as $m \ge 6$) and disjoint from
$s^{sum}_{p_4,j}$ for all unshifted cells on $s$. Hence $f^{sum}_4$ is injective on $D^n_{i+j=s}$.

\vspace{8pt}

\emph{Diff-diagonal direction, condition (1) of Lemma \ref{lem:inj-maj}.}
\hspace{1pt} \\
We compute $l^{n,p_4}_{i,j}$ on each of the four shifted diff regions (using $n = 2m = 3k$,
$m \ge 6$, $k \ge 4$):
\begin{itemize}[nosep]
  \item \emph{Region A ($1 + k \le j \le m - 3$ and $i \ge k$)}: here $p_4(j) = 2j + 2$, so
  \[
      l^{n,p_4}_{i,j} = (i + 2j + 2) \bmod n = (i + 2j + 2 - n) \bmod n.
  \]
  Since $i \in [k,n-1]$ and $j \in [k+1,m-3]$, we have
  \[
      0 < 4 = k + 2(k + 1) + 2 - 3k \le i + 2j + 2 - n \le (n - 1) + 2(m - 3) + 2 - n = n - 5 < n,
  \]
  so $l^{n,p_4}_{i,j} = i + 2j + 2 - n \ge 4 > 1$, and
  $0 < 3 \le i + 2j + 1 - n = l^{n,p_4}_{i,j} - 1 = f^{diff}_4(i,j) < l^{n,p_4}_{i,j}$.
  \item \emph{Region B ($j = m - 1 + k$ and $k \le i < 2k$)}: here $p_4(m-1+k) = 2k + 1$, so
  \[
      l^{n,p_4}_{i,m-1+k} = (i + 2k + 1) \bmod n = (i + 2k + 1 - 3k) \bmod n = (i - k + 1) \bmod n.
  \]
  Since $i \in [k,2k-1]$, we have $0 < 1 \le i - k + 1 \le k < n$, so
  $l^{n,p_4}_{i,m-1+k} = i - k + 1 \ge 1$, and
  $0 \le i - k = l^{n,p_4}_{i,m-1+k} - 1 = f^{diff}_4(i,m-1+k) < l^{n,p_4}_{i,m-1+k}$.
  \item \emph{Region C ($j = m - 1 + k$ and $2k \le i \le n - 1$)}: by the same reduction, for
  $i \in [2k,3k-1]$ we have $0 < k + 1 \le i - k + 1 \le 2k < n$, so
  $l^{n,p_4}_{i,m-1+k} = i - k + 1 \ge k + 1$.
  \begin{itemize}[nosep]
    \item If $n = 12$ ($k = 4$), $l^{n,p_4}_{i,m-1+k} = i - 3 \ge 5 > 4$, so
    $0 < 1 \le i - 7 = l^{n,p_4}_{i,m-1+k} - 4 = f^{diff}_4(i,m-1+k) < l^{n,p_4}_{i,m-1+k}$.
    \item If $n \ne 12$ ($n \ge 18$, so $k \ge 6$), $l^{n,p_4}_{i,m-1+k} = i - k + 1 \ge k + 1 \ge 7 > 5$,
    so $0 < 2 \le i - k - 4 = l^{n,p_4}_{i,m-1+k} - 5 = f^{diff}_4(i,m-1+k) < l^{n,p_4}_{i,m-1+k}$.
  \end{itemize}
  \item \emph{Region D ($m + k \le j \le n - 2$ and $i \ge k$)}: here $p_4(j) = 2(j - m + 1) + 1$, so
  \begin{align*}
      l^{n,p_4}_{i,j}
      &= (i + 2(j - m + 1) + 1) \bmod n \\
      &= (i + 2j - 2m + 3) \bmod n \\
      &= (i + 2j - n + 3) \bmod n \qquad (\text{since } 2m = n) \\
      &= (i + 2j - 2n + 3) \bmod n.
  \end{align*}
  Since $i \in [k,n-1]$ and $j \in [m+k,n-2]$ (with $2m + 3k = 2n$), we have
  \[
      0 < 3 = k + 2(m + k) - 2n + 3 \le i + 2j - 2n + 3 \le (n - 1) + 2(n - 2) - 2n + 3 = n - 2 < n,
  \]
  so $l^{n,p_4}_{i,j} = i + 2j - 2n + 3 \ge 3 > 2$, and
  $0 < 1 \le i + 2j - 2n + 1 = l^{n,p_4}_{i,j} - 2 = f^{diff}_4(i,j) < l^{n,p_4}_{i,j}$.
\end{itemize}
Off Regions A--D, $f^{diff}_4(i,j) = l^{n,p_4}_{i,j}$ holds by equality.

\vspace{8pt}

\emph{Diff-diagonal direction, condition (2) of Lemma \ref{lem:inj-maj}.}
\hspace{1pt} \\
Fix a diff-diagonal $d = i - j$. Because $3 \mid n$, $(d + s^{dif\!f}_{p_4,j}) \bmod n$ preserves
its residue class modulo~$3$:
\begin{itemize}[nosep]
  \item The unshifted part of the first main block $[1, m-3]$ is injective and lies in residue class
  $d + 2 \pmod 3$.
  \item The unshifted part of the second main block $[m-1, n-2]$ is injective and lies in residue
  class $d \pmod 3$.
  \item Substituting $i = d + j$ into the non-modular formulas from condition~(1), the shifted
  Regions A--D all land in the third residue class $d + 1 \pmod 3$ with pairwise disjoint offsets
  $f^{diff}_4(i,j) - d$:
  \begin{itemize}[nosep]
    \item On Region A ($1 + k \le j \le m - 3$, $i \ge k$):
    $f^{diff}_4(i,j) = d + 3(j - k) + 1$, whose offset $3(j - k) + 1$ is strictly increasing in $j$
    and lies in $[4, 3(m - 3 - k) + 1] = [4, m - 8]$ (empty for $n = 12$).
    \item On Region D ($m + k \le j \le n - 2$, $i \ge k$):
    $f^{diff}_4(i,j) = d + 3(j - m - k) + m + 1$, whose offset $3(j - m - k) + m + 1$ is strictly
    increasing in $j$ and lies in $[m + 1, n - 5]$, strictly above $[4, m - 8]$.
    \item On Region B ($j = m - 1 + k$, $k \le i < 2k$):
    $f^{diff}_4(i,m-1+k) = i - k = d + m - 1$, with offset $m - 1 \notin [4, m - 8] \cup [m + 1, n - 5]$.
    \item On Region C ($j = m - 1 + k$, $2k \le i \le n - 1$): for $n = 12$ ($m = 6, k = 4$),
    Region A is empty and $f^{diff}_4(i,9) = i - 7 = d + 2 \in [1,4]$ on $d \in [-1,2]$, which does
    not collide with $j_1 = 1$ ($l^{12,p_4}_{d+1,1} = d + 5$) or Region D ($f^{diff}_4(d+10,10) = d + 7$);
    for $n \ge 18$ ($k \ge 6$), $f^{diff}_4(i,m-1+k) = i - k - 4 = d + m - 5$, with offset
    $m - 5 \notin [4, m - 8] \cup \{m - 1\} \cup [m + 1, n - 5]$.
  \end{itemize}
  \item Finally, for the three boundary pins $j \in \{0, m-2, n-1\}$:
  \begin{itemize}[nosep]
    \item $j = m - 2$ has $l^{n,p_4}_{i,m-2} = i = d + m - 2 \equiv d + 1 \pmod 3$, with offset
    $m - 2 \notin [4, m - 8] \cup \{m - 5, m - 1\} \cup [m + 1, n - 5]$ (and for $n = 12$, offset
    $4 \ne 2$).
    \item $j = 0$ has $l^{n,p_4}_{i,0} = (i + n - 2) \bmod n$, which equals $d - 2$ for $i \ge 2$
    (offset $-2 < 4$) and $i + n - 2 \in \{n-2, n-1\}$ for $i \in \{0,1\}$ ($d \in \{0,1\}$, where
    all cells in $[1,n-1]$ with $i - j = d$ have values $< n - 2$).
    \item $j = n - 1$ has $l^{n,p_4}_{i,n-1} = (i + 2) \bmod n$, which equals $i + 2 = d + n + 1$
    for $i \le n - 3$ (offset $n + 1 > n - 5$) and $i + 2 - n \in \{0,1\}$ for
    $i \in \{n-2, n-1\}$ ($d \in \{-1,0\}$, where all other cells on $D^n_{i-j=d}$ have values $\ge 2$).
  \end{itemize}
\end{itemize}
Hence $f^{diff}_4$ is injective on $D^n_{i-j=d}$.

\end{proof}

\subsection[Small cases]{Small cases $n \in \{8, 9, 15, 21\}$}

The four families above cover all $n \ge 4$ with $n \notin \{4, 6\}$ except the four values
$n \in \{8, 9, 15, 21\}$, which fall below the thresholds $n \ge 10$ of family $F_2$ and
$n \ge 27$ of family $F_3$. For each of these we define $p_n$ and the two majorants
explicitly and verify incrementality by direct finite enumeration (the same enumeration has
been formalized in the Lean~4 companion \texttt{IncrementalNCCoins.lean}, see the GitHub
repository).

\vspace{8pt}

\noindent In what follows, $\val(i,j) := l^{n,p_n}_{i,j}$ for the value of the $(i,j)$-cell
of the specific square.

\paragraph{Case $n = 8$.} $p_8 = (0, 4, 1, 3, 6, 2, 7, 5)$; that is
$p_8(0) = 0, p_8(1) = 4, \dots, p_8(7) = 5$.
\begin{align*}
    f^{sum}_8(i,j) &=
    \begin{cases}
        i + 1, & j = 3 \text{ and } i < 4, \\
        3,     & j = 3 \text{ and } i = 4, \\
        \val(i,j), & \text{otherwise};
    \end{cases} \\
    f^{diff}_8(i,j) &=
    \begin{cases}
        i - 5, & j = 6 \text{ and } i \in \{5,6\}, \\
        5,     & j = 6 \text{ and } i = 7, \\
        \val(i,j), & \text{otherwise}.
    \end{cases}
\end{align*}

\paragraph{Case $n = 9$.} $p_9 = (0, 6, 3, 1, 8, 5, 2, 4, 7)$.
\begin{align*}
    f^{sum}_9(i,j) &=
    \begin{cases}
        i + 4,  & j = 5 \text{ and } i < 3, \\
        2,      & j = 5 \text{ and } i = 3, \\
        i,      & j = 6 \text{ and } i < 2, \\
        2i - 3, & j = 6 \text{ and } i \in \{2,3\}, \\
        \val(i,j), & \text{otherwise};
    \end{cases} \\
    f^{diff}_9(i,j) &= \val(i,j).
\end{align*}

\paragraph{Case $n = 15$.}
$p_{15} = (0, 3, 6, 9, 12, 8, 5, 2, 13, 1, 4, 7, 10, 14, 11)$.
\begin{align*}
    f^{sum}_{15}(i,j) &= \val(i,j); \\
    f^{diff}_{15}(i,j) &=
    \begin{cases}
        5,     & j = 13 \text{ and } i = 10, \\
        i - 8, & j = 13 \text{ and } 11 \le i \le 14, \\
        i - 5, & j = 14 \text{ and } i \in \{5,6\}, \\
        i - 6, & j = 14 \text{ and } 7 \le i \le 10, \\
        2,     & j = 14 \text{ and } i \in \{11,12,14\}, \\
        1,     & j = 14 \text{ and } i = 13, \\
        \val(i,j), & \text{otherwise}.
    \end{cases}
\end{align*}

\paragraph{Case $n = 21$.}
$p_{21} = (0, 18, 15, 12, 9, 6, 3, 7, 4, 1, 14, 17, 10, 16, 2, 5, 11, 8, 20, 13, 19)$.
\begin{align*}
    f^{sum}_{21}(i,j) &=
    \begin{cases}
        i + 6, & j = 7 \text{ and } i \le 12, \\
        13,    & j = 7 \text{ and } i = 13, \\
        i + 3, & j = 8 \text{ and } i \le 7, \\
        i + 2, & j = 8 \text{ and } 8 \le i \le 10, \\
        i + 1, & j = 8 \text{ and } i \in \{11,13\}, \\
        9,     & j = 8 \text{ and } i = 12, \\
        i,     & j = 9 \text{ and } i \le 7, \\
        i - 2, & j = 9 \text{ and } i \in \{8,9,10,12,13\}, \\
        7,     & j = 9 \text{ and } i = 11, \\
        i + 1, & j = 14 \text{ and } i \le 5, \\
        6,     & j = 14 \text{ and } i = 6, \\
        i + 1, & j = 14 \text{ and } 7 \le i \le 9, \\
        \val(i,j), & \text{otherwise};
    \end{cases} \\
    f^{diff}_{21}(i,j) &=
    \begin{cases}
        i - 13, & j = 18 \text{ and } 16 \le i \le 20, \\
        i - 14, & j = 19 \text{ and } 14 \le i \le 16, \\
        i - 17, & j = 19 \text{ and } 17 \le i \le 20, \\
        \val(i,j), & \text{otherwise}.
    \end{cases}
\end{align*}

\begin{lemma}\label{lem:small}
\hspace{1pt} \\ For each $n \in \{8, 9, 15, 21\}$ with $p_n$, $f^{sum}_n$, $f^{diff}_n$ as above,
$L^{n,p_n}$ is an incremental cyclic Latin square.
\end{lemma}

\begin{proof}

\vspace{8pt}

\hspace{1pt} \\ By Lemma~\ref{lem:inj-maj} and direct finite enumeration of majorization and
injectivity on $[0,n-1]^2$ for each $n \in \{8, 9, 15, 21\}$ (the enumeration is $O(n^3)$ per case
and has been checked in the Lean~4 companion).

\end{proof}

\subsection[Existence of Incremental Cyclic Latin Squares]{Existence of Incremental Cyclic Latin Squares}

\begin{theorem}\label{thm:main2}
\hspace{1pt} \\ $\forall n \in \mathbb{N}, \; n \ge 4, \; n \notin \{4, 6\} \; \exists L^{n,p}$ that
is an \emph{incremental} cyclic Latin square.
\end{theorem}

\begin{proof}

\vspace{8pt}

\hspace{1pt} \\ Let $n \in \mathbb{N}$ with $n \ge 4$ and $n \notin \{4, 6\}$. Partitioning $n$ by
its residue modulo~$6$:
\begin{itemize}
\item If $n = 1, 5 \bmod 6$, then $n \ge 5$, so $L^{n,p_1}$ is incremental by Lemma~\ref{lem:fam1}.
\item If $n = 2, 4 \bmod 6$, then since $n \notin \{4, 6\}$, either $n = 8$ (covered by
Lemma~\ref{lem:small}) or $n \ge 10$ (covered by Lemma~\ref{lem:fam2}).
\item If $n = 3 \bmod 6$, then either $n \in \{9, 15, 21\}$ (covered by Lemma~\ref{lem:small}) or
$n \ge 27$ (covered by Lemma~\ref{lem:fam3}).
\item If $n = 0 \bmod 6$, then since $n \ne 6$, we have $n \ge 12$ (covered by
Lemma~\ref{lem:fam4}).
\end{itemize}
In all cases, there exists an incremental cyclic Latin square $L^{n,p}$.

\end{proof}

\section[Non-Cyclic Cases and Combined Existence]{Non-Cyclic Cases $n \in \{4, 6\}$ and Combined Existence for All $n \ge 4$}

For $n \in \{4, 6\}$, no cyclic Latin square is incremental, but explicit non-cyclic incremental
Latin squares $L^4$ and $L^6$ exist:
\[
    L^4 :=
    \begin{pmatrix}
        2 & 0 & 1 & 3 \\
        1 & 3 & 2 & 0 \\
        0 & 2 & 3 & 1 \\
        3 & 1 & 0 & 2
    \end{pmatrix},
    \qquad
    L^6 :=
    \begin{pmatrix}
        1 & 0 & 2 & 3 & 5 & 4 \\
        3 & 1 & 4 & 0 & 2 & 5 \\
        4 & 3 & 5 & 2 & 1 & 0 \\
        0 & 2 & 1 & 5 & 4 & 3 \\
        2 & 5 & 0 & 4 & 3 & 1 \\
        5 & 4 & 3 & 1 & 0 & 2
    \end{pmatrix}.
\]

\vspace{8pt}

\begin{lemma}\label{lem:L4L6}
\hspace{1pt} \\ $L^4$ and $L^6$ are incremental Latin squares.
\end{lemma}

\begin{proof}

\vspace{8pt}

\hspace{1pt} \\ Each row and column of $L^4$ is a permutation of $[0,3]$, each row and column of
$L^6$ is a permutation of $[0,5]$, and direct finite verification of Definition~\ref{def:is} on all
sum-diagonals and diff-diagonals shows $|\,D^{n,k}_{i+j=s}\,| \le k + 1$ and
$|\,D^{n,k}_{i-j=d}\,| \le k + 1$ for every $k \in [0,n-1]$.

\end{proof}

\vspace{8pt}

\begin{theorem}\label{thm:main3}
\hspace{1pt} \\ $\forall n \in \mathbb{N}, \; n \ge 4 \; \exists L^n$ that is an \emph{incremental}
Latin square.
\end{theorem}

\begin{proof}

\vspace{8pt}

\hspace{1pt} \\ If $n \in \{4, 6\}$, Lemma~\ref{lem:L4L6} provides the incremental Latin square
$L^4$ or $L^6$. If $n \notin \{4, 6\}$, Theorem~\ref{thm:main2} provides the incremental cyclic
Latin square $L^{n,p}$.

\end{proof}

\vspace{8pt}

\begin{theorem}\label{thm:main4}
\hspace{1pt} \\ $\forall n \in \mathbb{N}, \; n \ge 4 \; \exists L^n$ that is an \emph{incremental}
Latin square representing $n + 1$ solutions of $(n,c)$-coins problems \emph{simultaneously} for
$c \in [0,n]$ as $S^n_c = \{L^n_{i,j} \mid l^n_{i,j} < c\}$.
\end{theorem}

\begin{proof}

\vspace{8pt}

\hspace{1pt} \\ By Theorem~\ref{thm:main3}, there exists an incremental Latin square $L^n$, and by
Theorem~\ref{thm:main1}, $L^n$ represents a solution $S^n_c$ of the $(n,c)$-coins problem for every
$c \in [0,n]$.

\end{proof}

\vspace{8pt}

\hspace{1pt} \\ The complete proofs of Theorems~\ref{thm:main1}, \ref{thm:main2}, \ref{thm:main3},
and \ref{thm:main4}, including the finite enumerations of Lemmas~\ref{lem:small} and \ref{lem:L4L6},
