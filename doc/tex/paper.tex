\title{\textbf{
  Cyclic Incremental Latin $n \times n$ Squares \\ 
  encoding \\
  Generalized $(n,c)$-Coins Squares for $0 \le c \le n$
}}
\author{Marc Duponcheel and Luc Duponcheel
}

\date{\today}

\begin{document}

\maketitle

\begin{abstract}

\noindent For every natural number $n \notin \{0, 2, 3, 4, 6\}$, there exists a permutation 
$q : \mathrm{Fin}\,n \to \mathrm{Fin}\,n$ such that
$L_{i,j} = (i + q(j)) \bmod n$ is an \emph{Cyclic Incremental Latin Square}, encoding $(n,c)$-Coins
Squares for $0 \le c \le n$.

\vspace{12pt}

An $(n,c)$-Coins Square for $0 \le 1 \le n$ corresponds to a solution of $n$-Queens Problem.

\vspace{12pt}

\noindent We have an existence proof in {\tt Lean} and construction code in {\tt Haskell},
{\tt Julia} and {\tt Lean}.

\vspace{12pt}

\noindent See e.g. \href{https://mathoverflow.net/questions/510135/placing-n-cdot-m-coins-on-an-n-times-n-board-generalizing-the-n-queens-p}{MO\hspace{-0.5pt}\#\hspace{-0.5pt}510135}, 
\href{https://puzzling.stackexchange.com/questions/138939/place-as-many-queens-as-you-can}{PSE\hspace{-0.5pt}\#\hspace{-0.5pt}138939},
\href{https://mathoverflow.net/questions/516225/universal-board-encoding-placement-of-n-cdot-m-coins-on-an-n-times-n-board}{MO\hspace{-0.5pt}\#\hspace{-0.5pt}516225} and
\href{https://github.com/NofanTasi/incremental-nc-coins-problem}{GitHub}.

\vspace{12pt}

\noindent The proof uses the \emph{Injective-Majorization Principle} which, in combination with the
\emph{Pigeon Hole Principle}, replaces intensive checking inequalities on each diagonal to the
existence of a single function $f(i,j) \le L_{i,j}$ which is injective on every diagonal.

\vspace{12pt}

\noindent The proof uses four families covering almost all natural number modulo~$6$ classes
\begin{itemize}
\item for $n \equiv 1,5 \pmod 6$ ($n \ge 1$)
\item for $n \equiv 2,4 \pmod 6$ ($n \ge 10$)
\item for $n \equiv 0 \pmod 6$ ($n \ge 12$)
\item for $n \equiv 3 \pmod 6$ ($n \ge 27$)
\end{itemize}

\vspace{12pt}

\noindent For all four families a Cyclic Incremental Latin Square solution can be constructed. Constructing
a Cyclic Incremental Latin Square solution for the four not covered natural numbers, $n \in \{8, 9, 15, 21\}$,
is handled separately. Likewise, constructing a (non-Cyclic) Incremental Latin Square solution for the two
remaining natural numbers, $n \in \{4, 6\}$, is also handled separately.

\end{abstract}

\section{Cyclic Incremental Latin Squares}

\subsection{From $(n,c)$-Coins Squares to Cyclic Incremental Latin Squares}

$(n,c)$-Coins Squares are subsets $S_c \subseteq \{0, \dots, n-1\}^2$ of $nc$ cells of an $n \times n$
square such that every row, column, sum-diagonal ($i + j = s$), and difference-diagonal
($i - j = d$) contains at most $c$ cells of $S_c$.

\vspace{12pt}

Clearly, for $c = 1$, they correspond to solutions of $n$-Queens Problems.

\vspace{12pt}

The paper of Le, Li and Wang~\cite{LLW1990} constructs $S_c$'s separately for each $c$.

\vspace{12 pt}

\subsection{The Conjecture}

Every permutation $q : \mathrm{Fin}\,n \to \mathrm{Fin}\,n$ defines a \emph{Cyclic Latin Square}:
\[
  L_{i,j} = (i + q(j)) \bmod n \qquad (0 \le i, j < n).
\]

\vspace{12 pt}

The conjecture is that for all $n > 0$ except $n \in \{2, 3, 4, 6\}$, there exists a column
permutation $q$ such that $L_{i,j} = (i + q(j)) \bmod n$ is a Cyclic Incremental Latin Square solutions.
For $n \in \{2, 3\}$, the $n$-queens problem has no solution, and, for $n \in \{4, 6\}$, exhaustive
finite search confirms that no Cyclic Incremental Latin Square solutions satisfies the diagonal bounds.

\begin{theorem}[Cyclic Incremental Latin Square Conjecture]\label{thm:main}
For every integer $n > 0$ and $n \notin \{2, 3, 4, 6\}$, there exists a bijection
$q : \mathrm{Fin}\,n \to \mathrm{Fin}\,n$ such that the Cyclic Incremental Latin Square
\[
  L_{i,j} = (i + q(j)) \bmod n
\]
satisfies:

\vspace{12pt}

\begin{enumerate}[nosep,label=(\roman*)]
  \item For every $i \in \mathrm{Fin}\,n$, the row map $j \mapsto L_{i,j}$ is bijective.
  \item For every $j \in \mathrm{Fin}\,n$, the column map $i \mapsto L_{i,j}$ is bijective.
  \item For every $0 \le k < n$ and every $s \in \mathbb{N}$,
  \[
    |\{(i,j) \in \mathrm{Fin}\,n \times \mathrm{Fin}\,n \mid i + j = s \;\land\; L_{i,j} \le k\}| \le k + 1.
  \]
  \item For every $0 \le k < n$ and every $d \in \mathbb{Z}$,
  \[
    |\{(i,j) \in \mathrm{Fin}\,n \times \mathrm{Fin}\,n \mid i - j = d \;\land\; L_{i,j} \le k\}| \le k + 1.
  \]
\end{enumerate}
\end{theorem}

\section{Injective Majorization}

Instead of direct cardinality counting of
  \[
    |\{(i,j) \in \mathrm{Fin}\,n \times \mathrm{Fin}\,n \mid i + j = s \;\land\; L_{i,j} \le k\}| \le k + 1.
  \]
and
  \[
    |\{(i,j) \in \mathrm{Fin}\,n \times \mathrm{Fin}\,n \mid i - j = d \;\land\; L_{i,j} \le k\}| \le k + 1.
  \]
which would computationally explode for large $n$, we use a simple combinatorial principle that
completely eliminates cardinality counting and reduces the problem for \emph{all} thresholds
$0 \le k < n$ simultaneously to constructing a single appropriate injection.

\subsection{The Injective-Majorization Principle}

Let $D \subseteq \mathrm{Fin}\,n \times \mathrm{Fin}\,n$ be any diagonal 
(either a sum-diagonal $i + j = s$ or a difference-diagonal $i - j = d$), and let 
$v(i,j) = (i + q(j)) \bmod n \in \{0, \dots, n-1\}$ be the square value at cell $(i,j)$.

\begin{lemma}[Injective-Majorization Principle] \label{lem:inj-maj}
If there exists a function $f : \mathbb{N} \times \mathbb{N} \to \mathbb{N}$ such that:

\vspace{12pt}

\begin{enumerate}[nosep,label=(\arabic*)]
  \item Majorization: $f(i,j) \le v(i,j)$ for all $0 \le i, j < n$,
  \item Diagonal Injectivity: $f$ is injective along $D$.
\end{enumerate}

\vspace{12pt}

Then for every threshold $k \in \mathbb{N}$, the set $D_{\le k} = \{(i,j) \in D \mid v(i,j) \le k\}$
has cardinality at most $k + 1$.
\end{lemma}

\begin{proof}
For every cell $(i,j) \in D_{\le k}$, condition~(1) gives
\[
  0 \le f(i,j) \le v(i,j) \le k,
\]
so $f(i,j) \in \{0, \dots, k\}$. By condition~(2), the restriction of $f$ to $D_{\le k}$ is an
injective map into $\{0, \dots, k\}$. By the Pigeonhole Principle,
\[
  |D_{\le k}| \le |\{0, \dots, k\}| = k + 1. \qedhere
\]
\end{proof}

\vspace{12pt}

\begin{remark}[Geometric Meaning of $f$]
Along a diagonal $D$, the square values $v(i,j)$ are \emph{almost} distinct, except for a small
number of pairs $(i_1, j_1), (i_2, j_2) \in D$ caused by jumps between piecewise-linear
blocks of $q(j)$ or modular wrap-around at $n$. Whenever two cells on $D$ share the same value
$v(i_1, j_1) = v(i_2, j_2) = v$, one of the two cells is allowed to keep its value, say 
$f(i_1, j_1) = v$ while the other is changed to an unused value $f(i_2, j_2) < v$ that does not
appear anywhere else along $D$. Because $f(i_2, j_2) < v(i_2, j_2)$ on the changed cell, the
changed cell steals a slot in $\{0, \dots, f(i_2, j_2)\} \subseteq \{0, \dots, v(i_2, j_2)\}$ that
was left vacant on $D$!
\end{remark}
